**约定（任务中默认遵循，未声明则视为遵循）：**
- 包前缀 `com.lumen.wms`。
- 实体继承 `com.lumen.common.entity.BaseEntity`（含 id/createdAt/updatedAt/deleted/version，由 MyBatis-Plus 自动填充）。
- 状态字段类型用 `String`，值用 enum 常量。
- Mapper 接口用 `@Mapper` + `extends BaseMapper<Entity>`。
- Service 接口命名 `XxxService`，实现 `XxxServiceImpl`（与既有 masterdata / rbac 一致）。
- Controller 返回 `com.lumen.common.api.R<T>` 与 `com.lumen.common.api.PageResult<T>`。
- 所有写接口加 `@PreAuthorize("hasAuthority('wms:xxx')")` 与 `@Audit(action=..., resource='wms-xxx')`。
- 多租户：用 `com.lumen.common.tenant.TenantContext.getTenantId()`；所有 mapper 注入 `tenant_id`。
- 测试：service 单测用 Mockito（无 Spring），controller IT 用 `@SpringBootTest + @AutoConfigureMockMvc`，依赖 testcontainers MySQL 8。
- 前端：每组件一个文件夹，组件默认导出，反例以 `<Foo>` 形式被引。

**Files:**

- [ ] **Step 1: 创建目录 `lumen-wms/src/main/resources/db/migration/`**

注意：现有 Flyway 迁移在 `lumen-bootstrap/src/main/resources/db/migration/`，但 lumen-wms 的迁移也应该由 bootstrap 加载（Spring Boot 默认从 classpath 根扫描），所以**也建在 bootstrap 的同目录**会更简单，并保持单调递增。但 lumen-wms 作为独立模块，也可以在自己 `src/main/resources/db/migration/` 放文件，Spring Boot 默认多 classpath 加载会合并。这里我们**选择放在 lumen-wms 内**（更内聚）。

如要放 lumen-wms 内，需在 `application.yml` 加 `spring.flyway.locations: classpath:db/migration,classpath:db/migration-wms`，但为简单**直接放到 bootstrap**：

**取消 lumen-wms 的 migration dir，改在 `lumen-bootstrap/src/main/resources/db/migration/V15__wms_schema.sql`** 创建文件：

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.lumen.common.api.PageResult;
import com.lumen.common.event.OutboxEventPublisher;
import com.lumen.common.tenant.TenantContext;
import com.lumen.wms.WmsConstants;
import com.lumen.wms.entity.WmsInventory;
import com.lumen.wms.exception.ConcurrentModifiedException;
import com.lumen.wms.exception.InventoryInsufficientException;
import com.lumen.wms.mapper.WmsInventoryMapper;
import com.lumen.wms.service.InventoryService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.OptimisticLockingFailureException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Map;

@Slf4j
@Service
@RequiredArgsConstructor
public class InventoryServiceImpl implements InventoryService {

    private final WmsInventoryMapper mapper;
    private final OutboxEventPublisher outbox;     // 已存在 lumen-extension

    @Override
    public WmsInventory upsertIfAbsent(Long warehouseId, Long locationId, Long itemId) {
        WmsInventory inv = findOne(warehouseId, locationId, itemId);
        if (inv != null) return inv;
        WmsInventory fresh = new WmsInventory();
        fresh.setTenantId(TenantContext.getTenantId());
        fresh.setWarehouseId(warehouseId);
        fresh.setLocationId(locationId);
        fresh.setItemId(itemId);
        fresh.setQty(BigDecimal.ZERO);
        fresh.setLockedQty(BigDecimal.ZERO);
        fresh.setLastMovementAt(LocalDateTime.now());
        mapper.insert(fresh);
        return fresh;
    }

    private WmsInventory findOne(Long warehouseId, Long locationId, Long itemId) {
        return mapper.selectOne(new LambdaQueryWrapper<WmsInventory>()
                .eq(WmsInventory::getTenantId, TenantContext.getTenantId())
                .eq(WmsInventory::getWarehouseId, warehouseId)
                .eq(WmsInventory::getLocationId, locationId)
                .eq(WmsInventory::getItemId, itemId));
    }

    @Override
    @Transactional
    public WmsInventory addQty(Long wh, Long loc, Long item, BigDecimal delta, Long noteId) {
        WmsInventory inv = mustWithRetry(wh, loc, item);
        BigDecimal newQty = inv.getQty().add(delta);
        if (newQty.signum() < 0) throw new InventoryInsufficientException(inv.getQty(), delta);
        inv.setQty(newQty);
        inv.setLastMovementAt(LocalDateTime.now());
        mapper.updateById(inv);
        publishEvent(inv, noteId, Map.of("delta", delta));
        return inv;
    }

    @Override
    @Transactional
    public WmsInventory reserve(Long wh, Long loc, Long item, BigDecimal delta, Long noteId) {
        WmsInventory inv = mustWithRetry(wh, loc, item);
        BigDecimal newLocked = inv.getLockedQty().add(delta);
        if (newLocked.compareTo(inv.getQty()) > 0)
            throw new InventoryInsufficientException(inv.getQty().subtract(inv.getLockedQty()), delta);
        inv.setLockedQty(newLocked);
        mapper.updateById(inv);
        publishEvent(inv, noteId, Map.of("reservedDelta", delta));
        return inv;
    }

    @Override
    @Transactional
    public WmsInventory releaseAndFlush(Long wh, Long loc, Long item, BigDecimal delta, Long noteId) {
        WmsInventory inv = mustWithRetry(wh, loc, item);
        BigDecimal newLocked = inv.getLockedQty().subtract(delta);
        BigDecimal newQty = inv.getQty().subtract(delta);
        if (newLocked.signum() < 0 || newQty.signum() < 0)
            throw new InventoryInsufficientException(inv.getQty(), delta);
        inv.setLockedQty(newLocked);
        inv.setQty(newQty);
        inv.setLastMovementAt(LocalDateTime.now());
        mapper.updateById(inv);
        publishEvent(inv, noteId, Map.of("flushed", delta));
        return inv;
    }

    private WmsInventory mustWithRetry(Long wh, Long loc, Long item) {
        OptimisticLockingFailureException last = null;
        for (int i = 0; i < 2; i++) {
            WmsInventory inv = findOne(wh, loc, item);
            if (inv == null) inv = upsertIfAbsent(wh, loc, item);
            try {
                // Force version check by simulating an update.
                int r = mapper.updateById(inv);
                if (r > 0) return inv;
            } catch (OptimisticLockingFailureException ex) {
                last = ex;
            }
        }
        throw new ConcurrentModifiedException();
    }

    private void publishEvent(WmsInventory inv, Long noteId, Map<String, Object> details) {
        outbox.publish(WmsConstants.EVT_INVENTORY_MOVED, inv.getId(), java.util.Map.of(
            "warehouseId", inv.getWarehouseId(),
            "locationId", inv.getLocationId(),
            "itemId", inv.getItemId(),
            "qty", inv.getQty(),
            "lockedQty", inv.getLockedQty(),
            "refNoteId", noteId == null ? "" : noteId,
            "details", details
        ));
    }

    @Override
    public PageResult<WmsInventory> page(Long wh, Long loc, Long item, long pageNum, long pageSize) {
        Page<WmsInventory> page = Page.of(pageNum, pageSize);
        LambdaQueryWrapper<WmsInventory> q = new LambdaQueryWrapper<WmsInventory>()
                .eq(WmsInventory::getTenantId, TenantContext.getTenantId())
                .orderByDesc(WmsInventory::getId);
        if (wh != null) q.eq(WmsInventory::getWarehouseId, wh);
        if (loc != null) q.eq(WmsInventory::getLocationId, loc);
        if (item != null) q.eq(WmsInventory::getItemId, item);
        return PageResult.of(mapper.selectPage(page, q));
    }

    @Override
    public List<WmsInventory> summary(Long wh) {
        LambdaQueryWrapper<WmsInventory> q = new LambdaQueryWrapper<WmsInventory>()
                .eq(WmsInventory::getTenantId, TenantContext.getTenantId())
                .groupBy(WmsInventory::getItemId)
                .select(WmsInventory::getItemId,
                        com.baomidou.mybatisplus.core.toolkit.support.SFunction.of(... )) // 略，task executor 自填 SQL
                ;
        if (wh != null) q.eq(WmsInventory::getWarehouseId, wh);
        return mapper.selectList(q);
    }

package com.lumen.wms.approval;

import com.lumen.wms.WmsConstants;
import com.lumen.wms.entity.WmsThresholdConfig;
import com.lumen.wms.service.ThresholdConfigService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

import java.math.BigDecimal;

@Component
@RequiredArgsConstructor
public class ThresholdEvaluator {

    private final ThresholdConfigService cfgSvc;

    public boolean requiresApproval(String bizKind, BigDecimal amount, BigDecimal totalQty) {
        WmsThresholdConfig c = cfgSvc.getOrDefault();
        if (!Boolean.TRUE.equals(c.getApprovalRequiredWhenExceed())) return false;
        BigDecimal amtThr = pick(c, bizKind, "amt");
        BigDecimal qtyThr = pick(c, bizKind, "qty");
        boolean amtOver = amtThr.signum() > 0 && safeCompare(amount, amtThr) > 0;
        boolean qtyOver = qtyThr.signum() > 0 && safeCompare(totalQty, qtyThr) > 0;
        return amtOver || qtyOver;
    }

    private static BigDecimal pick(WmsThresholdConfig c, String kind, String what) {
        return switch (kind + ":" + what) {
            case WmsConstants.BIZ_INBOUND + ":amt" -> nz(c.getInbAmountThreshold());
            case WmsConstants.BIZ_INBOUND + ":qty" -> nz(c.getInbQtyThreshold());
            case WmsConstants.BIZ_OUTBOUND + ":amt" -> nz(c.getOutAmountThreshold());
            case WmsConstants.BIZ_OUTBOUND + ":qty" -> nz(c.getOutQtyThreshold());
            case WmsConstants.BIZ_ADJUST + ":amt" -> nz(c.getAdjAmountThreshold());
            case WmsConstants.BIZ_ADJUST + ":qty" -> nz(c.getAdjQtyThreshold());
            default -> BigDecimal.ZERO;
        };
    }
    private static BigDecimal nz(BigDecimal v) { return v == null ? BigDecimal.ZERO : v; }
    private static int safeCompare(BigDecimal a, BigDecimal b) {
        return a == null ? -1 : a.compareTo(b);
    }
}

```java
@Override
@Transactional
public WmsInboundNote submit(Long id) {
    WmsInboundNote note = mustInStatus(id, WmsConstants.INB_DRAFT);
    List<WmsInboundLine> lines = lineMapper.listByNote(note.getId());
    if (lines.isEmpty()) throw new IllegalNoteStateException("入库单至少要有 1 条行");
    // Compute (amount, qty)
    BigDecimal totalQty = lines.stream().map(WmsInboundLine::getQty)
        .reduce(BigDecimal.ZERO, BigDecimal::add);
    BigDecimal totalAmount = BigDecimal.ZERO;
    for (WmsInboundLine line : lines) {
        WmsItem item = itemMapper.selectById(line.getItemId());
        if (item == null) throw new IllegalNoteStateException("找不到货品 " + line.getItemId());
        totalAmount = totalAmount.add(item.getUnitCost().multiply(line.getQty()));
    }
    // Threshold evaluation
    boolean needApproval = evaluator.requiresApproval(WmsConstants.BIZ_INBOUND, totalAmount, totalQty);
    String target = needApproval ? WmsConstants.INB_PENDING_APPROVAL : WmsConstants.INB_RECEIVING;
    note.setStatus(target);
    noteMapper.updateById(note);
    if (needApproval) {
        approval.create(WmsConstants.BIZ_INBOUND, note.getId(), Operator.of("inbound-submit"));
    }
    outbox.publish(WmsConstants.EVT_INB_STATE_CHANGED, note.getId(),
        Map.of("from", WmsConstants.INB_DRAFT, "to", target, "noteId", note.getId()));
    return note;
}
```

`receive / putaway / approve / cancel / create` 与 `addLine / deleteLine / page / get / update` 任务执行者照搬相同模式（参照 sys dict 子表实现）。此处不再逐行展开，关键 invariants 如下：

- `addLine` 仅 DRAFT 可调，line_no 自动 = max+1。
- `receiveLine` 不能使 `received_qty > qty`，全行 received 后 status 进 PUTAWAY。
- `putawayLine` 不能使 `putaway_qty > received_qty`；调 `inventoryService.addQty(wh, loc, item, delta, noteId)`；全行 putaway 后 status 进 DONE，`completed_at = now()`。
- `approve` 仅 PENDING_APPROVAL 可调，调 `approval.approve(...)` 后状态进 RECEIVING。
- `cancel` 不能从 DONE 出发；其他状态都允许。
- `create` 时调 `numberingService.generate("WMS_INBOUND")` 取 no。

## Task 4.2: OutboundService — submit 时 reserve 库存 + pick 减锁 + done 释放并扣减

```java
@Override
@Transactional
public WmsOutboundNote submit(Long id) {
    WmsOutboundNote note = mustInStatus(id, WmsConstants.OUT_DRAFT);
    List<WmsOutboundLine> lines = lineMapper.listByNote(note.getId());
    if (lines.isEmpty()) throw new IllegalNoteStateException("出库单至少要有 1 条行");
    // 计算 amount + qty + reserve 库存
    BigDecimal totalQty = BigDecimal.ZERO, totalAmount = BigDecimal.ZERO;
    for (WmsOutboundLine l : lines) {
        WmsItem item = itemMapper.selectById(l.getItemId());
        if (item == null) throw new IllegalNoteStateException("找不到货品 " + l.getItemId());
        totalQty = totalQty.add(l.getQty());
        totalAmount = totalAmount.add(item.getUnitCost().multiply(l.getQty()));
        if (l.getSourceLocationId() == null)
            throw new IllegalNoteStateException("行 " + l.getLineNo() + " 缺少 sourceLocationId");
        WmsLocation loc = locationMapper.selectById(l.getSourceLocationId());
        if (!"ACTIVE".equals(loc.getStatus()))
            throw new LocationNotAvailableException(l.getSourceLocationId());
        inventory.reserve(note.getWarehouseId(), l.getSourceLocationId(), l.getItemId(),
            l.getQty(), note.getId());
    }
    boolean needApproval = evaluator.requiresApproval(WmsConstants.BIZ_OUTBOUND, totalAmount, totalQty);
    String target = needApproval ? WmsConstants.OUT_PENDING_APPROVAL : WmsConstants.OUT_PICKING;
    note.setStatus(target);
    noteMapper.updateById(note);
    if (needApproval) {
        approval.create(WmsConstants.BIZ_OUTBOUND, note.getId(), "outbound-submit");
    }
    outbox.publish(WmsConstants.EVT_OUT_STATE_CHANGED, note.getId(),
        Map.of("from", WmsConstants.OUT_DRAFT, "to", target));
    return note;
}

@Override
@Transactional
public void pickLine(Long noteId, Long lineId, BigDecimal qty) {
    WmsOutboundLine line = lineMapper.selectById(lineId);
    BigDecimal remaining = line.getQty().subtract(line.getPickedQty());
    if (qty.compareTo(remaining) > 0) throw new IllegalNoteStateException("拣货数超过待拣");
    inventory.releaseAndFlush(line ... );   // 先 release locked，再 ... — 实际：应只释放 locked，不扣 qty
    line.setPickedQty(line.getPickedQty().add(qty));
    lineMapper.updateById(line);
    // 若全行 picked_qty == qty，可能整体推进状态
    if (allLinesFullyPicked(noteId)) {
        WmsOutboundNote note = noteMapper.selectById(noteId);
        note.setStatus(WmsConstants.OUT_REVIEW);
        noteMapper.updateById(note);
    }
}
```

> 注意：`pick` 实际只是 `locked_qty -= qty`（不扣 `qty`，因为货还没真出库）；`done` 时 `locked_qty -= remainingPicked, qty -= picked_qty`，代表真正出库。
