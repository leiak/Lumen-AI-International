```java
import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
...

    /**
     * 按 id 取收 / 付款流水.
     *
     * <p>显式带 tenant_id + deleted = 0 —— 租户隔离的**运行期**兜底是
     * MyBatis-Plus 的 TenantLineInnerInterceptor, 但那依赖 TenantContext 的隐式行为:
     * 一旦这张表被加进豁免名单, 或该方法在没有租户上下文 (定时任务 / MQ 消费) 的
     * 场景下被复用, 就会真的读到别家的流水。显式谓词是纵深防御。
     *
     * <p>deleted 只能手写 —— 实体没有 deleted 字段 (DDL 有), 用不了 @TableLogic。
     *
     * <p>错误码用专用的 PAYMENT_RECORD_NOT_FOUND (404_007), 与
     * BillingPaymentRecordController#get 的 javadoc 一致; 早先抛的是通用
     * NOT_FOUND (404_005) —— 那个码至今仍被 ReconciliationServiceImpl.get 使用着,
     * 语义太泛, 表达不了"付款记录不存在或不属于本租户"这个具体语义。
     */
    @Override
    public BillingPaymentRecord get(Long tenantId, Long id) {
        BillingPaymentRecord r = mapper.selectOne(new LambdaQueryWrapper<BillingPaymentRecord>()
                .eq(BillingPaymentRecord::getId, id)
                .eq(BillingPaymentRecord::getTenantId, tenantId)
                .apply("deleted = 0"));
        if (r == null) throw new FinanceException(FinanceErrorCode.PAYMENT_RECORD_NOT_FOUND);
        return r;
    }
```

`list` 改成:

```java
    @Override
    public List<BillingPaymentRecord> list(Long tenantId, String period, String direction, int page, int size) {
        LambdaQueryWrapper<BillingPaymentRecord> q = new LambdaQueryWrapper<>();
        // tenant_id / deleted 与 MerchantBillingViewServiceImpl.paymentScope 保持一致 ——
        // 商家视图与管理视图看到的是同一批行, 不能一个看得到软删行另一个看不到。
        q.eq(BillingPaymentRecord::getTenantId, tenantId)
         .apply("deleted = 0");
        if (period != null) q.eq(BillingPaymentRecord::getPeriod, period);
        if (direction != null) q.eq(BillingPaymentRecord::getDirection, direction);
        q.orderByDesc(BillingPaymentRecord::getId);
        return mapper.selectPage(new Page<>(page, size), q).getRecords();
    }
```

> `LambdaQueryWrapper.eq(field, null)` 会生成 `field = null` 而不是跳过 —— 所以 `period` / `direction` 仍用 `if` 守卫。

```java
    @Override
    public List<BillingAdjustment> list(Long tenantId, String status, Boolean disputed, int page, int size) {
        int pn = page < 1 ? 1 : page;
        int ps = size < 1 ? 20 : Math.min(size, 200);
        Page<BillingAdjustment> pg = Page.of(pn, ps);

        // 4.5C-2.1 — disputed 过滤用「Java 侧 id 集 + in / notIn」, 不用 SQL 子查询:
        // TenantLineInnerInterceptor 对相关子查询的处理没验证过, 而 id 集这条路
        // 4.5C-2 的 ownedAdjustmentIds 已在生产跑通。
        Set<Long> disputedIds = null;
        if (disputed != null) {
            disputedIds = disputedAdjustmentIds(tenantId);
            if (Boolean.TRUE.equals(disputed) && disputedIds.isEmpty()) {
                // 一条异议都没有 → 直接空页。绝不能生成 id IN () 那种非法 SQL。
                return List.of();
            }
            if (Boolean.FALSE.equals(disputed) && disputedIds.isEmpty()) {
                // 全都没有异议 → 不加任何过滤条件 (notIn 空集 = 全部排除, 语义反了)。
                disputedIds = null;
            }
        }

        LambdaQueryWrapper<BillingAdjustment> q = new LambdaQueryWrapper<BillingAdjustment>()
            .eq(BillingAdjustment::getTenantId, tenantId)
            .eq(status != null && !status.isBlank(),
                BillingAdjustment::getStatus, status);
        if (disputedIds != null) {
            if (Boolean.TRUE.equals(disputed)) {
                q.in(BillingAdjustment::getId, disputedIds);
            } else {
                q.notIn(BillingAdjustment::getId, disputedIds);
            }
        }
        q.orderByDesc(BillingAdjustment::getId);

        Page<BillingAdjustment> result = adjustmentMapper.selectPage(pg, q);
        List<BillingAdjustment> records = result.getRecords();
        if (records == null || records.isEmpty()) {
            return List.of();
        }
        List<BillingAdjustment> out = new ArrayList<>(records);
        attachDisputes(tenantId, out);
        return out;
    }
```

**(d) 加两个 private 辅助方法:**

```java
    /**
     * 4.5C-2.1 — 该租户下**有商家异议**的调账单 id 集。
     *
     * <p>deleted 只能手写: BillingReconciliation 实体没有 deleted 字段 (DDL 有)。
     */
    private Set<Long> disputedAdjustmentIds(Long tenantId) {
        return reconMapper.selectList(new LambdaQueryWrapper<BillingReconciliation>()
                        .select(BillingReconciliation::getAdjustmentId)
                        .eq(BillingReconciliation::getTenantId, tenantId)
                        .isNotNull(BillingReconciliation::getAdjustmentId)
                        .isNotNull(BillingReconciliation::getDisputeAt)
                        .apply("deleted = 0"))
                .stream()
                .map(BillingReconciliation::getAdjustmentId)
                .filter(Objects::nonNull)
                .collect(Collectors.toCollection(LinkedHashSet::new));
    }

    /**
     * 4.5C-2.1 — 把商家异议快照贴到调账单实体上 (瞬态字段, 不落库)。
     *
     * <p>权威来源是 t_reconciliation 的 dispute_* 三列, 这里是只读快照。
     *
     * <p>同一个 adjustment 被多条 recon 认领 (脏数据) 时取 <b>dispute_at 最新</b>的那条:
     * 查询已按 dispute_at DESC 排序, 先到先得即可。刻意不做"稳定挑一个"的兜底 ——
     * 那会把脏数据永久藏起来。
     */
    private void attachDisputes(Long tenantId, List<BillingAdjustment> adjustments) {
        if (adjustments == null || adjustments.isEmpty()) {
            return;
        }
        List<Long> ids = adjustments.stream()
                .map(BillingAdjustment::getId)
                .filter(Objects::nonNull)
                .toList();
        if (ids.isEmpty()) {
            return;
        }
        List<BillingReconciliation> recons = reconMapper.selectList(
                new LambdaQueryWrapper<BillingReconciliation>()
                        .eq(BillingReconciliation::getTenantId, tenantId)
                        .in(BillingReconciliation::getAdjustmentId, ids)
                        .isNotNull(BillingReconciliation::getDisputeAt)
                        .apply("deleted = 0")
                        .orderByDesc(BillingReconciliation::getDisputeAt));
        if (recons == null || recons.isEmpty()) {
            return;
        }
        Map<Long, BillingReconciliation> latest = new HashMap<>();
        for (BillingReconciliation r : recons) {
            latest.putIfAbsent(r.getAdjustmentId(), r);   // 已按 dispute_at DESC 排序 → 先到先得
        }
        for (BillingAdjustment adj : adjustments) {
            BillingReconciliation r = latest.get(adj.getId());
            if (r == null) {
                continue;    // [N13-4] 无异议 → 3 个字段保持 null
            }
            adj.setDisputeReason(r.getDisputeReason());
            adj.setDisputeAmount(r.getDisputeAmount());
            adj.setDisputeAt(r.getDisputeAt());
        }
    }
```

### R1 [重要] 「原样透传」这条核心声明没被锁住

现有 3 个用例传的 page/size 分别是 `3/5`、`2/50`、`4/10`, **全在 finance 的钳位
区间内** —— 所以 controller 里加 clamp (`Math.max(page,1)` / `Math.min(size,200)`)
照样全绿。

**失败场景**: 有人觉得"controller 侧也该兜一下", 把 `page` 截到 100 → 商家有
10,000 条付款流水时**永远看不到第 100 页之后** —— 正是本 task 要消灭的"只能看前
N 条"的翻版。而没有任何测试会红。

**改法**: 给 3 个 controller 各补一条**越界值**用例, 断言的是"原样"而不是
"clamp 后的值":

```java
    @Test

    void recon_doesNotClamp() {
        when(billingView.listReconciliations(anyLong(), anyLong(), isNull(), anyLong(), anyLong()))
                .thenReturn(MerchantPage.empty(0, 500));

        reconController.list(null, 0L, 500L);

        // 关键: 断言"原样", 而不是 clamp 后的值
        verify(billingView).listReconciliations(TENANT, CUSTOMER, null, 0L, 500L);
    }
```

invoice / payment 同理各一条 (`page=0, size=500`)。

### R2 [重要] `defaultValue` 零覆盖 —— 改成 50 没有任何测试变红

`@RequestParam(defaultValue=...)` 是 **Spring MVC 层**的, 直接调方法传的是显式
参数, 所以单测永远碰不到它。而前端 `portalBillingService.ts` 目前**不传**
`page`/`size` → 这个默认值就是商家当前唯一生效的每页行数。

**失败场景**: 把 `defaultValue = "20"` 改成 `"50"` → 商家门户首页每页从 20 行
静默变成 50 行, 无任何测试变红; 若前端分页器有硬编码的 20 (Task 5 落地时容易
发生), 后端 50 / 前端 20 → **商家会看到重复或漏掉的行**。

**改法** (二选一, 优先 MockMvc —— 它验的是 Spring 真实的绑定行为):

```java
    @Test
    @DisplayName("不传 page / size → service 收到 1 / 20 (默认值锁死, 改了这里就红)")
    void absentPagingParamsUseDefaultOneTwenty() throws Exception {
        MerchantBillingViewService svc = mock(MerchantBillingViewService.class);
        when(svc.listReconciliations(anyLong(), anyLong(), isNull(), anyLong(), anyLong()))
                .thenReturn(MerchantPage.empty(1, 20));
        MockMvc mockMvc = MockMvcBuilders.standaloneSetup(new PortalReconController(svc)).build();

        mockMvc.perform(get("/api/customer/recon")).andExpect(status().isOk());

        verify(svc).listReconciliations(anyLong(), anyLong(), isNull(), eq(1L), eq(20L));
    }
```

复审员已实测: 本仓 `MockMvcBuilders.standaloneSetup` 可用 (`spring-boot-starter-test`
自带 `spring-test`); standalone 下 `@PreAuthorize` **不生效**, 所以这个测试
**不能**用来做鉴权断言 (鉴权由 `PortalBillingControllerSecurityTest` 负责)。

若 MockMvc 走不通, 退路是反射读注解 (本仓 `-parameters` 已由
`spring-boot-starter-parent` 开启, `Parameter.getName()` 能拿到 `"page"`/`"size"`):

```java
    private static String defaultValueOf(Class<?> type, String method, String paramName) throws Exception {
        for (java.lang.reflect.Parameter p : type.getDeclaredMethod(method).getParameters()) {
            if (p.getName().equals(paramName)) {
                RequestParam rp = p.getAnnotation(RequestParam.class);
                return rp == null ? null : rp.defaultValue();
            }
        }
        throw new AssertionError(paramName + " not found");
    }
```

### R3 [重要] finance 侧 `clampPage` 的两个分支零覆盖

`MerchantBillingViewServiceImpl.clampPage` (`Math.min(Math.max(page,1), 1_000_000)`)
的 `page<1→1` 与 `page>1_000_000→1_000_000` **两个分支都没有测试** —— 既有测试只
覆盖了 size 的钳位 (`sizeZeroFallsBackTo20` / `sizeTooLargeIsCapped`), 所有
`listPayments` 调用**只传过 `page = 1`**。

**为什么算在本 task 头上**: 改动前门户把 `page` 写死成 1, 这条分支对商家
**不可达**; 是本 commit 让它**第一次对商家可达**。

**失败场景**: 有人"简化" `clampPage` (觉得 `MAX_PAGE` 这个上界多余, 删掉
`Math.min`) → 商家请求 `?page=9223372036854775807` → `Page.offset()` 不检查溢出,
offset 变负 → 拼出 `LIMIT -40, 20` 是 MySQL 语法错 → **整个请求 500**。现有测试
一条都不会红。

**改法**: 在 `lumen-parent/lumen-finance/src/test/java/com/lumen/finance/service/MerchantBillingViewServiceTest.java`
的「分页与钳位」段 (现有 `listPayments_sizeTooLargeIsCapped` 之后) 补两条:

```java
    @Test
    @DisplayName("page=0 → 生效 1")
    void listPayments_pageBelowOneFallsBackTo1() {
        when(paymentMapper.selectPage(any(), any())).thenAnswer(fillPage(List.of(), 0L));

        MerchantPage<MerchantDtos.PaymentView> out = service.listPayments(TENANT, ME, null, 0, 20);

        assertThat(out.page()).isEqualTo(1);
    }

    @Test
    @DisplayName("page=Long.MAX_VALUE → 截断到 1_000_000 (否则 offset 溢出成负 → LIMIT 负数 → 500)")
    void listPayments_pageTooLargeIsCapped() {
        when(paymentMapper.selectPage(any(), any())).thenAnswer(fillPage(List.of(), 0L));

        MerchantPage<MerchantDtos.PaymentView> out = service.listPayments(TENANT, ME, null, Long.MAX_VALUE, 20);

        assertThat(out.page()).isEqualTo(1_000_000);
    }
```

> `fillPage` / `TENANT` / `ME` 是该测试类里已有的辅助与常量, 直接用。

#### R3 的下界用例 —— 实施中发现的偏离 (2026-11-06)

本节原计划把下界用例写成 `listPayments_pageBelowOneFallsBackTo1`。**实施时做变异验证
发现那条是假绿**: 删掉 `clampPage` 的 `Math.max(page, MIN_PAGE)` 后它**照样绿**。

**原因**: MyBatis-Plus 3.5.7 的 `Page(long current, long size)` 构造器**自己**就把
`current <= 1` 归一成 1 —— 字节码 (javap) 实证: `current` 字段初值为 1, 且只在
`current > 1` 时才赋值 (`lload_1; lconst_1; lcmp; ifle`)。所以凡是走 MP `Page`
的路径 (`listPayments` / `listReconciliations` / `listInvoices`), **下界根本不可
观测** —— 删了 clamp 的下界, 结果一模一样。

**真正只靠 `clampPage` 兜底下界的是"归属集为空"那条分支** ——
`MerchantPage.empty(clampPage(page), clamp(size))` 压根不经过 MP 的 `Page`。

**所以下界用例放在 `listMyAdjustments` 的无归属分支上**:

```java
    @Test
    @DisplayName("page=0 → 生效 1 (listMyAdjustments 的无归属分支不经过 MP 的 Page, 看得见 clamp 的下界)")
    void listMyAdjustments_pageBelowOneFallsBackTo1() {
        // 归属集为空 → 直接回 MerchantPage.empty(clampPage(page), clamp(size))
        MerchantPage<MerchantDtos.AdjustmentView> out =
                service.listMyAdjustments(TENANT, ME, 0, 20);

        assertThat(out.page()).isEqualTo(1);
    }
```

> 计划原定的 `listPayments_pageBelowOneFallsBackTo1` **也保留** —— 它仍是一条有效的
> 行为断言 (只是它测不到 clamp 的下界, 测的是"走 MP 的路径时页码归一后也是 1")。
> 两条都在, 覆盖才完整。
>
> **教训记在案**: 给别人的实现配测试时, "看起来能覆盖"的路径未必真的能观测到被测
> 的分支 —— 依赖库的内部归一化会把差异吃掉。**必须做变异验证**才能确认用例有效。

### R4 [次要] javadoc 里的 clamp 描述漏了 `page` 的上界

3 个 controller 的 javadoc 都写 "clamp (size<=0→20, size>200→200, page<1→1)",
但 `clampPage` 还有 `page>1_000_000→1_000_000` (service 第 82 行 `MAX_PAGE`,
注释里专门解释过为什么必须有)。**行为没错** (service 兜住了), 但文档少了一条
真实契约 —— 读 controller javadoc 的人会以为 `?page=99999999999` 原样进 DB。

**改法**: 3 处统一改成
`clamp (size<=0→20, size>200→200, page<1→1, page>1000000→1000000)`。

### R5 [次要] 没断言"返回值原样透传"

`when(...)` 打桩返回了 `MerchantPage.empty(3, 5)`, 但没有任何断言检查 controller
的返回值。**失败场景**: 有人把 `R.ok(billingView.listReconciliations(...))` 改成
`R.ok(MerchantPage.empty(1, 20))` (手滑) → 商家拿到空列表 → 测试全绿。

**改法**: 3 个已有用例各加两行:

```java
        R<MerchantPage<ReconView>> r = reconController.list(null, 3L, 5L);
        assertThat(r.getData().page()).isEqualTo(3);
        assertThat(r.getData().size()).isEqualTo(5);
```

(`R` 是 `com.lumen.common.api.R`, `getData()` 取数据。)

### 已决定**不做**的一项

**默认值 `20` 散在 5 个地方** (3 个 controller 的 `defaultValue="20"` +
`PortalAdjustmentController` 的 `size == null ? 20L : size` +
`MerchantBillingViewServiceImpl.DEFAULT_PAGE_SIZE = 20L`), 复审员建议抽成公共
常量。**协调者决定不做**, 理由:

1. `@RequestParam(defaultValue=...)` 要求**编译期 String 常量**, 而 service 侧的
   `DEFAULT_PAGE_SIZE` 是 `long` —— 两者类型不同, 无法真正共用一个常量; 强行统一
   只能再加一个 `String` 副本, 反而多一处。
2. 统一要改 `PortalAdjustmentController` 与 `MerchantBillingViewServiceImpl`, 跨
   2 个模块, 与本 task "只改 controller 的参数透传" 的范围不符。
3. 复审员担心的失败场景 ("产品改默认值只改一处 → 门户与后台口径分叉") 的实际危害
   被 **R2 压掉大半**: 默认值被测试锁住后, 改它一定会有测试变红, 不会静默。

**记在案**: 这是已知的风格不一致, 不阻塞本轮。

|---|---|---|
| M3b | 3 个 controller **只**加 `Math.min(page, 100L)` | ❌ **9 条全绿** → R6 |
| M5 | `listInvoices` 整道 clamp 删掉 | ❌ **40 条全绿** → R7 |
| M6' | `listReconciliations` 整道 clamp 删掉 | ❌ **40 条全绿** → R7 |

### R6 [重要] 越界值用例漏掉 page 的**上界**

R1 原来传的是 `page = 0` —— `Math.min(0, 100) == 0`, 与期望值相同 → verify 照样通过。
**没有任何单个 page 值能同时抓住 `Math.max` 和 `Math.min` 两种 clamp。**

改法: 3 条 `*_doesNotClamp` 各打三个方向 —— page 下界 (`page=0`) / page 上界
(`page=2_000_000`, 大于 finance 的 `MAX_PAGE=1_000_000`) / size 上界 (`size=500`),
R9 再补 size 下界 (`size=0`)。

### R7 [重要] `listInvoices` / `listReconciliations` 的 **clamp 调用点**零覆盖

`clamp` / `clampPage` 是**共享的私有静态方法** (函数逻辑 R3 已测全), 但**调用点是
4 份独立的代码**。4b15d26 正是让 `listInvoices` / `listReconciliations` 这两个调用点
**第一次对商家可达**的那次提交 —— 改动前门户把 page/size 写死成 1/20。

失败场景: 商家 `?page=9223372036854775807` → `Page.of(Long.MAX_VALUE, 20)` →
`Page.offset()` = `(MAX-1)*20` **long 溢出成 -40** → SQL 拼出 `LIMIT -40, 20` →
MySQL 语法错 → **整个请求 500**。

### R8 [次要] MockMvc 用例里身份参数用的是 `anyLong()`

改成 `eq(TENANT), eq(CUSTOMER)`。审查者用 V9 (篡改 ThreadLocal 里的 tenant) 实测
**9 条全红** (含 3 条 MockMvc) → 证明 MockMvc 路径真读到了身份, `eq(TENANT)` 是真断言。

### R9 [次要] size 的下界 (`size=0`) 没覆盖

`size=500` 抓得住 `Math.min(size, N)` (N<500), 抓不住 `Math.max(size, 1)`。

### R10 [重要] 第 5 个 clamp 调用点: `listMyAdjustments` **有数据**分支

**位置**: `MerchantBillingViewServiceImpl.java:481`
`adjustmentMapper.selectPage(Page.of(clampPage(page), clamp(size)), ...)`

**审查者实测**: 只把这一处换成 `Page.of(page, size)` → **42 条全绿**。
(对照: `listPayments` 的 L271 换成同样的东西 → **4 败**。)

**为什么漏**: `listMyAdjustments` 的两条分支是**互斥**的 —— 归属集为空 → 提前 return
`MerchantPage.empty(clampPage(page), clamp(size))` (不经过 MP 的 `Page`); 归属集非空
→ 走 `selectPage(Page.of(...))`。而此前所有 `listMyAdjustments` 的测试都传
`page=1, size=20` → clamp 是**恒等变换**, 看不见; 唯一用了越界值 (`size=1000`) 的
`emptyPage_reportsSameEffectiveSizeAsNonEmptyPage` 走的恰好是**归属集为空**那条分支,
压根没进 `selectPage`。

**失败场景**: 商家 `GET /api/customer/adjustment?page=9223372036854775807&size=100000`
→ `PortalAdjustmentController` 原样透传 (它本来就暴露 page/size, 比本 task 还早) →
商家**确实有**调账单 → 归属集非空 → 若这里丢了 `clampPage` → offset long 溢出成负
→ `LIMIT 负数` → MySQL 语法错 → **整个请求 500**。

**注意这个 500 只在"商家确实有调账单"时发生** —— 归属集为空时走提前 return 那条分支,
那里 clamp 还在。所以它是"绝大多数商家"会踩到的那种。

**改法**: 补 `listMyAdjustments_withData_clampsPaging` (跳 1 认领一行
`adjustmentId=31` → 归属集非空 → 进 `selectPage` 分支)。

**至此 5 个 clamp 调用点全部有独立覆盖**:

| 行号 | 方法 / 分支 | 覆盖用例 |
|---|---|---|
| L98 | `listReconciliations` | `listReconciliations_clampsPaging` |
| L203 | `listInvoices` | `listInvoices_clampsPaging` |
| L271 | `listPayments` | 既有 3 条 |
| L479 | `listMyAdjustments` 空归属分支 | `listMyAdjustments_pageBelowOneFallsBackTo1` + `emptyPage_reportsSameEffectiveSizeAsNonEmptyPage` |
| L481 | `listMyAdjustments` 有数据分支 | `listMyAdjustments_withData_clampsPaging` |

### 这一轮学到的两件事 (记在案)

1. **共享的私有方法 ≠ 共享的覆盖**。`clamp` / `clampPage` 的**函数逻辑**被一条
   用例测全, 不代表**每个调用点**都被测到 —— 调用点是 4 份独立的代码, 删掉其中
   一份的调用, 测试一条都不会红。**补测试时要按调用点枚举, 不是按函数枚举。**
2. **恒等变换是不可观测的**。测试传 `page=1, size=20` 时, clamp 是恒等变换,
   加不加它结果都一样 —— 这类"看起来覆盖了"的路径其实什么都没锁。要让钳位可观测,
   必须传**越界值**, 而且必须走到**那一条具体的分支** (空归属 vs 非空归属)。

```ts
/**
 * 4.5C-2.1 — 服务端分页 hook (缺口 #9)。
 *
 * 商家门户的 4 个列表页都是服务端分页: 后端返回**当前页**的 records + 全量
 * total, 前端不再自己切页。
 *
 * 之所以用 hook 而不是各页各写一遍: 「改页大小要回到第 1 页」这条规则很容易漏,
 * 漏了的表现是 —— 商家在第 5 页把页大小从 20 改成 10, 结果停在一个空页上,
 * 看起来像数据丢了。
 */

import { useCallback, useState } from 'react';
import type { TablePaginationConfig } from 'antd';
import { zhCN } from '../locale/zh-CN';

/** 页大小选项 —— 上限 100, 不得超过后端 clamp 上限 (200)。 */
export const PAGE_SIZE_OPTIONS = ['10', '20', '50', '100'];

export interface ServerPagination {
  page: number;
  size: number;
  /** 直接展开给 <Table pagination={...}> */
  pagination: TablePaginationConfig;
}

/**
 * @param total 后端返回的总条数 —— 喂给 antd 分页器, 商家才看得到"一共多少条"。
 */
export function useServerPagination(total: number, initialSize = 20): ServerPagination {
  const [page, setPage] = useState(1);
  const [size, setSize] = useState(initialSize);

  const onChange = useCallback(
    (nextPage: number, nextSize: number) => {
      setSize(nextSize);
      // 页大小变了 → 回到第 1 页。旧的页码在新的页大小下可能已经越界,
      // 停在那个页码上商家会看到一个空页, 像是数据丢了。
      setPage(nextSize === size ? nextPage : 1);
    },
    [size],
  );

  const pagination: TablePaginationConfig = {
    current: page,
    pageSize: size,
    total,
    showSizeChanger: true,
    pageSizeOptions: PAGE_SIZE_OPTIONS,
    showTotal: (t: number) => zhCN.billing.totalCount(t),
    onChange,
  };

  return { page, size, pagination };
}
```

## 附录 B — 已知的坑 (给实现者)

1. **antd 服务端分页**: `dataSource.length` **必须** `<= pageSize`, 否则 antd 会切片并告警
   (`antd/es/table/InternalTable.js:228-246`)。后端 clamp 200 + 前端只请求当前页 → 满足。
2. **MP 的 lambda 缓存**: 纯 Mockito 单测里 `getSqlSegment()` 会抛
   "can not find lambda cache for this entity" —— 必须先
   `TableInfoHelper.initTableInfo(assistant, XxxEntity.class)`。参考
   `MerchantBillingViewServiceTest.initMybatisPlusLambdaCache`。
3. **MP 的参数 map 是懒生成的**: 先 `getSqlSegment()` 再读 `getParamNameValuePairs()`,
   否则拿到空 map, 断言变成恒真。
4. **`BillingReconciliation` / `BillingPaymentRecord` 实体没有 `deleted` 字段** ——
   软删过滤必须手写 `.apply("deleted = 0")`。
5. **`BillingAdjustment` 连 `deleted` 列都没有** —— 调账单侧**不能**写 `deleted` 过滤
   (会 `Unknown column 'deleted'` → 500)。
6. **`AdjustmentServiceImpl` 用显式构造器** —— 加 `reconMapper` 后 `AdjustmentServiceTest.setUp`
   必须同步改。
7. **`@TableField(exist = false)` 不影响 JSON 序列化** —— admin-web 能直接拿到这 3 个字段。
8. **别改 V1..V40** —— 本切片**不写任何迁移**。
9. **`npm run lint` 在本仓是坏的** —— 不进验收标准。
10. **不要跑全量回归** (`mvn test` at root) —— 只跑本文件里列出的命令。
