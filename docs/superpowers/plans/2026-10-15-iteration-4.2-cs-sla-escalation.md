# Iteration 4.2 — 客服中心 SLA 与升级 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 4.1 异常工单基础上落地客服中心（lumen-cs）—— V24 migration（4 表）+ 3 级 SLA 升级（customer_service → cs_supervisor → cs_manager）+ 业务时间算法 + 跨模块 outbox 事件。

**Architecture:** 新增 maven 模块 `lumen-cs`（与 lumen-dispatch 同级，第 9 子模块）。lumen-cs 依赖 lumen-dispatch（事件契约 `DeliveryExceptionStateChangedEvent` + 只读视图 `WmsDeliveryExceptionView`），反向依赖仍禁止；通过 4.1.x 落地的 `DomainEventTypeRegistrar` SPI + outbox 双向事件流（dispatch 发 DeliveryExceptionStateChangedEvent → cs 订阅；cs 发 ExceptionEscalatedEvent → dispatch 订阅）实现模块解耦。SLA 计时用 5min `@Scheduled` 轮询 + `t_sla_event.UNIQUE KEY` 防并发重复升级。

**Tech Stack:**
- 后端：Spring Boot 3.2.5 + MyBatis-Plus 3.5.7 + JUnit5 + AssertJ + Mockito + Flyway 9.x + testcontainers
- 前端：Vite 5 + React 18 + TS 5 + Antd 5 + @ant-design/pro-components 2.7 + Vitest 1 + RTL + MSW 2
- 状态机：复用 4.1 既有 `t_state_transition` + `StateTransitionService`（4.2 仅 V25 migration 修 guard 表达式）
- 跨模块事件：复用 4.1.x 的 `DomainEventTypeRegistrar` + `EventBus.publish`

**Spec:** `docs/superpowers/specs/2026-10-15-iteration-4.2-cs-sla-escalation-design.md`

**Delivery:** 6 个 PR，每个 PR 各自可 `mvn verify` + `npm run build`。完成后 `mvn -pl lumen-bootstrap verify -Dit.test='!OutboxRetryIT'` 集成测试目标 **81+** 通过（已有 69 + 新增 12）。

  CHECK (end_time > start_time)

  version INT NOT NULL DEFAULT 0,
  UNIQUE KEY uk_tenant_date (tenant_id, holiday_date, deleted)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

                throw new CsException(CsErrorCode.DUPLICATE_SLA_CONFIG);

            throw e;
        }

    public GetResponse update(Long tenantId, Long id, UpdateRequest req) {

        row.setResponseMinutes(req.responseMinutes());
        row.setResolveMinutes(req.resolveMinutes());
        row.setEscalateToLevel(req.escalateToLevel());
        mapper.updateById(row);
        return toDto(row);
    }

    @Override
    public void delete(Long tenantId, Long id) {
        SlaConfig row = mustFind(tenantId, id);
        mapper.deleteById(row.getId());
    }

    public GetResponse get(Long tenantId, Long id) {

    public ListResponse list(Long tenantId, ListRequest req) {

package com.lumen.cs.calculator;

import org.springframework.stereotype.Component;

import java.time.*;
import java.util.List;

@Component
public class SlaCalculator {

    /**
     * 计算 [from, to] 之间的「业务分钟数」。
     * 算法：按天迭代；首末日按时间戳比例算；节假日/非工作日跳过。
     */
    public long businessMinutes(TenantCalendar cal, LocalDateTime from, LocalDateTime to) {
        if (to.isBefore(from) || to.isEqual(from)) return 0;
        long totalMinutes = 0;
        LocalDate cursor = from.toLocalDate();
        LocalDate endDate = to.toLocalDate();
        while (!cursor.isAfter(endDate)) {
            // Skip holidays and non-work weekdays
            int weekday = cursor.getDayOfWeek().getValue();
            List<TenantCalendar.TimeRange> ranges = cal.weekdayRanges().get(weekday);
            if (ranges == null || cal.holidays().contains(cursor)) {
                cursor = cursor.plusDays(1);
                continue;
            }
            for (TenantCalendar.TimeRange r : ranges) {
                LocalDateTime segStart = maxOf(from, LocalDateTime.of(cursor, r.start()));
                LocalDateTime segEnd = minOf(to, LocalDateTime.of(cursor, r.end()));
                if (segEnd.isAfter(segStart)) {
                    totalMinutes += Duration.between(segStart, segEnd).toMinutes();
                }
            }
            cursor = cursor.plusDays(1);
        }
        return totalMinutes;
    }

    private LocalDateTime maxOf(LocalDateTime a, LocalDateTime b) {
        return a.isAfter(b) ? a : b;
    }

    private LocalDateTime minOf(LocalDateTime a, LocalDateTime b) {
        return a.isBefore(b) ? a : b;
    }
}

package com.lumen.cs.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.lumen.cs.dto.EscalationDtos.*;
import com.lumen.cs.entity.SlaEvent;
import com.lumen.cs.mapper.SlaEventMapper;
import com.lumen.cs.service.SlaEventService;
import lombok.RequiredArgsConstructor;
import org.springframework.dao.DuplicateKeyException;
import org.springframework.stereotype.Service;

import java.time.LocalDateTime;
import java.util.List;

@Service
@RequiredArgsConstructor
public class SlaEventServiceImpl implements SlaEventService {

    private final SlaEventMapper mapper;

    @Override
    public SlaEventDto record(Long tenantId, Long exceptionId, String eventType,
                              String currentLevel, String severity, LocalDateTime occurredAt,
                              String payloadJson) {
        SlaEvent row = new SlaEvent();
        row.setTenantId(tenantId);
        row.setExceptionId(exceptionId);
        row.setEventType(eventType);
        row.setCurrentLevel(currentLevel);
        row.setSeverity(severity);
        row.setOccurredAt(occurredAt);
        row.setPayload(payloadJson);
        try {
            mapper.insert(row);
        } catch (DuplicateKeyException e) {
            // 幂等：UNIQUE KEY 命中说明已记录，跳过
        }
        return toDto(row);
    }

    @Override
    public List<SlaEventDto> listByException(Long tenantId, Long exceptionId) {
        return mapper.selectList(new LambdaQueryWrapper<SlaEvent>()
                .eq(SlaEvent::getTenantId, tenantId)
                .eq(SlaEvent::getExceptionId, exceptionId)
                .orderByAsc(SlaEvent::getOccurredAt))
            .stream().map(this::toDto).toList();
    }

    @Override
    public SlaEventDto getLatestResponse(Long tenantId, Long exceptionId) {
        SlaEvent row = mapper.selectOne(new LambdaQueryWrapper<SlaEvent>()
                .eq(SlaEvent::getTenantId, tenantId)
                .eq(SlaEvent::getExceptionId, exceptionId)
                .eq(SlaEvent::getEventType, "L1_RESPONDED")
                .orderByDesc(SlaEvent::getOccurredAt)
                .last("LIMIT 1"));
        return row == null ? null : toDto(row);
    }

    private SlaEventDto toDto(SlaEvent row) {
        return new SlaEventDto(
            row.getId(), row.getTenantId(), row.getExceptionId(), row.getEventType(),
            row.getCurrentLevel(), row.getSeverity(), row.getOccurredAt(), row.getPayload()
        );
    }
}

package com.lumen.cs.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.lumen.cs.calculator.SlaCalculator;
import com.lumen.cs.calculator.TenantCalendar;
import com.lumen.cs.dto.EscalationDtos.*;
import com.lumen.cs.entity.SlaConfig;
import com.lumen.cs.entity.SlaEvent;
import com.lumen.cs.entity.WmsDeliveryExceptionView;
import com.lumen.cs.exception.CsErrorCode;
import com.lumen.cs.exception.CsException;
import com.lumen.cs.mapper.SlaConfigMapper;
import com.lumen.cs.mapper.SlaEventMapper;
import com.lumen.cs.mapper.WmsDeliveryExceptionViewMapper;
import com.lumen.cs.service.BusinessCalendarService;
import com.lumen.cs.service.EscalationService;
import com.lumen.cs.service.HolidayService;
import com.lumen.cs.service.SlaEventService;
import com.lumen.extension.outbox.EventBus;
import com.lumen.dispatch.event.ExceptionEscalatedEvent;  // 由 PR5 引入
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import java.time.LocalDateTime;
import java.util.*;

@Service
@RequiredArgsConstructor
@Slf4j
public class EscalationServiceImpl implements EscalationService {

    private final SlaConfigMapper configMapper;
    private final SlaEventMapper eventMapper;
    private final WmsDeliveryExceptionViewMapper exceptionMapper;
    private final SlaCalculator calculator;
    private final BusinessCalendarService calendarService;
    private final HolidayService holidayService;
    private final SlaEventService slaEventService;
    private final EventBus eventBus;
    private final ObjectMapper objectMapper = new ObjectMapper();

    @Override
    public void scan() {
        for (long tenantId : listActiveTenants()) {
            try {
                scanTenant(tenantId);
            } catch (Exception e) {
                log.error("SLA scan failed for tenant {}", tenantId, e);
            }
        }
    }

    private void scanTenant(long tenantId) {
        TenantCalendar cal = loadCalendar(tenantId);
        List<WmsDeliveryExceptionView> open = exceptionMapper.selectOpenByTenant(tenantId);
        for (WmsDeliveryExceptionView ex : open) {
            checkOne(cal, ex);
        }
    }

    private void checkOne(TenantCalendar cal, WmsDeliveryExceptionView ex) {
        String severity = deriveSeverity(ex.getType());
        String currentLevel = ex.getHandler() == null ? "L1" : inferLevel(ex.getHandler());
        SlaConfig config = findConfig(ex.getTenantId(), severity, currentLevel);
        if (config == null || config.getEscalateToLevel() == null) return;

        LocalDateTime start = resolveStart(ex);
        long elapsed = calculator.businessMinutes(cal, start, LocalDateTime.now());
        if (elapsed <= config.getResolveMinutes()) return;

        // 1. 写 t_sla_event (ESCALATED, current_level = currentLevel) — UNIQUE KEY 防并发
        Map<String, Object> payload = new LinkedHashMap<>();
        payload.put("businessMinutesElapsed", elapsed);
        payload.put("resolveTargetMinutes", config.getResolveMinutes());
        try {
            slaEventService.record(ex.getTenantId(), ex.getId(), "ESCALATED", currentLevel, severity,
                LocalDateTime.now(), objectMapper.writeValueAsString(payload));
        } catch (JsonProcessingException e) {
            throw new CsException(CsErrorCode.SEVERITY_MAPPING_MISSING, e.getMessage());
        }

        // 2. 发 outbox ExceptionEscalatedEvent
        String newHandler = pickUserForLevel(ex.getTenantId(), config.getEscalateToLevel());
        eventBus.publish(new ExceptionEscalatedEvent(
            ex.getId(), ex.getTenantId(),
            currentLevel, config.getEscalateToLevel(),
            severity, elapsed, newHandler
        ));
    }

    private String deriveSeverity(String type) {
        return switch (type) {
            case "DAMAGE" -> "HIGH";
            case "OVERTIME", "REJECTED" -> "MEDIUM";
            case "ADDR_ERR" -> "LOW";
            default -> throw new CsException(CsErrorCode.SEVERITY_MAPPING_MISSING, type);
        };
    }

    private String inferLevel(String handlerUsername) {
        // handler 格式: "username" 包含 role code（简化规则）
        if (handlerUsername.contains("cs_manager")) return "L3";
        if (handlerUsername.contains("cs_supervisor")) return "L2";
        return "L1";
    }

    private LocalDateTime resolveStart(WmsDeliveryExceptionView ex) {
        SlaEvent responded = eventMapper.selectOne(new LambdaQueryWrapper<SlaEvent>()
            .eq(SlaEvent::getExceptionId, ex.getId())
            .eq(SlaEvent::getEventType, "L1_RESPONDED")
            .orderByDesc(SlaEvent::getOccurredAt)
            .last("LIMIT 1"));
        return responded != null ? responded.getOccurredAt() : ex.getCreatedAt();
    }

    private SlaConfig findConfig(Long tenantId, String severity, String level) {
        return configMapper.selectOne(new LambdaQueryWrapper<SlaConfig>()
            .eq(SlaConfig::getTenantId, tenantId)
            .eq(SlaConfig::getSeverity, severity)
            .eq(SlaConfig::getLevel, level)
            .last("LIMIT 1"));
    }

    private TenantCalendar loadCalendar(long tenantId) {
        // 4.2.0 简化：cal + holidays 由 service 提供
        // 由 PR2 的 BusinessCalendarService.load(tenantId) 引入
        // TODO PR4 finalization: replace stub
        return new TenantCalendar(Map.of(), Set.of());
    }

    private String pickUserForLevel(long tenantId, String level) {
        // 简化：返回形如 "system_{tenant}_{level}"，4.2 后续接入 user-role 表
        return "system_" + tenantId + "_" + level;
    }

    private List<Long> listActiveTenants() {
        // 简化：只扫 tenant 1 (dev)；4.2 后续接入 sys_tenant
        return List.of(1L);
    }
}

package com.lumen.cs.listener;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.lumen.cs.service.SlaEventService;
import com.lumen.dispatch.event.DeliveryExceptionStateChangedEvent;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.event.EventListener;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Component;

import java.util.Map;

@Component
@RequiredArgsConstructor
@Slf4j
public class DeliveryExceptionStateChangedEventListener {

    private final SlaEventService slaEventService;
    private final ObjectMapper om = new ObjectMapper();

    @EventListener
    @Async("outboxExecutor")
    public void onExceptionTransitioned(DeliveryExceptionStateChangedEvent e) {
        String type = switch (e.getToState()) {
            case "OPEN"          -> "L1_RESPONSE_DUE";
            case "INVESTIGATING" -> "L1_RESPONDED";
            case "RESOLVED"      -> "RESOLVED";
            case "CLOSED"        -> "RESOLVED";  // 视作终止
            case "CANCELLED"     -> "CANCELLED";
            default -> null;
        };
        if (type == null) return;
        String payload;
        try {
            payload = om.writeValueAsString(Map.of(
                "fromState", e.getFromState() == null ? "" : e.getFromState(),
                "action", e.getAction() == null ? "" : e.getAction()
            ));
        } catch (JsonProcessingException ex) {
            payload = "{}";
        }
        // severity 由 EscalationService 派生；这里传 null 让 SlaEventService 兼容
        slaEventService.record(e.getTenantId(), e.getExceptionId(), type,
            inferLevel(e.getToState()), "HIGH" /*placeholder*/, e.getOccurredAt(), payload);
    }

    private String inferLevel(String toState) {
        return switch (toState) {
            case "OPEN" -> "L1";
            default -> "L1";  // 简化；EscalationService 真正决定 level
        };
    }
}

package com.lumen.dispatch.listener;

import com.lumen.dispatch.entity.WmsDeliveryException;
import com.lumen.dispatch.event.ExceptionEscalatedEvent;
import com.lumen.dispatch.mapper.WmsDeliveryExceptionMapper;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.event.EventListener;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Component;

@Component
@RequiredArgsConstructor
@Slf4j
public class ExceptionEscalatedEventListener {

    private final WmsDeliveryExceptionMapper mapper;

    @EventListener
    @Async("outboxExecutor")
    public void onEscalated(ExceptionEscalatedEvent e) {
        WmsDeliveryException ex = mapper.selectById(e.getExceptionId());
        if (ex == null) {
            log.warn("Escalation event for unknown exception {}", e.getExceptionId());
            return;
        }
        ex.setHandler(e.getNewHandlerUsername());
        mapper.updateById(ex);
        log.info("Exception {} escalated {} -> {}, handler set to {}",
            e.getExceptionId(), e.getFromLevel(), e.getToLevel(), e.getNewHandlerUsername());
    }
}

interface Props {

export default function SlaCountdownTag({ elapsedMinutes, targetMinutes }: Props) {
  const ratio = elapsedMinutes / targetMinutes;
  const color = ratio >= 1 ? 'red' : ratio >= 0.8 ? 'orange' : 'green';
  const remaining = targetMinutes - elapsedMinutes;
  const label = remaining >= 0
    ? `剩余 ${remaining}min / ${targetMinutes}min`
    : `超期 ${-remaining}min / ${targetMinutes}min`;
  return <Tag color={color}>{label}</Tag>;
}
