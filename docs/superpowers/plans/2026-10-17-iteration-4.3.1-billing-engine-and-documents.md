# Iteration 4.3.1 — lumen-finance 计费引擎 + 单据基础 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新建 `lumen-finance` Maven 子模块（第 10 子模块），实现基于 Aviator 表达式沙箱的可配置计费引擎，订阅上游业务事件（4.1 派车 + 3.0 WMS 入库出库）后自动写"暂估 + 应收/应付单据"，支持 1 级审批 + 2 个 FE 报表页。

**Architecture:** 事件驱动 (lumen-finance 订阅 lumen-extension outbox 事件，不反向依赖 lumen-dispatch / lumen-wms) + 表达式沙箱 (Aviator 白名单变量 + 100ms 超时 + 8 逃逸测试) + 2 阶段计费 (accrual OPEN → LOCKED → invoice DRAFT) + 1 级审批 (复用 lumen-dispatch 的 `StateMachineEngine`)。

**Tech Stack:** Spring Boot 3.2.5 + MyBatis-Plus 3.5.7 + Flyway 9.x + Aviator 5.x (表达式引擎) + JUnit 5 + Mockito + SpringBootTest + Awaitility。

```java
package com.lumen.finance.exception;

import com.lumen.common.error.ErrorCode;
import lombok.Getter;

/**
 * Finance 模块业务错误码 (Iteration 4.3.1).
 *
 * <p>错误码区间：60001..60099 (5 位编码, 与 dispatch / wms 风格一致).
 * 4.3.1 范围约 12 个 code, 4.3.2/4.3.3 续号.
 *
 * <p>HTTP 状态码由 {@link #httpStatus} 字段决定 (lumen-common GlobalExceptionHandler 自动映射).
 */
@Getter
public enum FinanceErrorCode implements ErrorCode {
    RULE_NOT_FOUND                    (404_001, "Rule not found",                           404),
    RULE_ITEM_NOT_FOUND               (404_002, "Rule item not found",                      404),
    ACCRUAL_NOT_FOUND                 (404_003, "Accrual not found",                        404),
    INVOICE_NOT_FOUND                 (404_004, "Invoice not found",                        404),
    EXPRESSION_INVALID                (400_001, "Expression invalid",                       400),
    EXPRESSION_TIMEOUT                (400_002, "Expression evaluation timeout",            400),
    EXPRESSION_UNKNOWN_VARIABLE       (400_003, "Expression references unknown variable",   400),
    DUPLICATE_RULE                    (409_001, "Rule for (tenant, business_type, direction, counterparty_type, effective_from) already exists", 409),
    INVOICE_NO_CONFLICT               (409_002, "Invoice number generation conflict after 3 retries", 409),
    INVALID_STATE_TRANSITION          (409_003, "Invalid state transition for billing invoice", 409),
    RULE_NOT_ACTIVE                   (400_004, "Rule is not in ACTIVE status",             400),
    EXPRESSION_SANDBOX_VIOLATION      (400_005, "Expression violates sandbox policy",       400);

    private final int code;
    private final String message;
    private final int httpStatus;

    FinanceErrorCode(int code, String message, int httpStatus) {
        this.code = code;
        this.message = message;
        this.httpStatus = httpStatus;
    }

    @Override
    public int httpStatus() { return httpStatus; }
}
```

```java
package com.lumen.finance.engine;

import com.googlecode.aviator.AviatorEvaluator;
import com.googlecode.aviator.AviatorEvaluatorOptions;
import com.googlecode.aviator.Expression;
import com.googlecode.aviator.Options;
import com.googlecode.aviator.runtime.type.AviatorRuntimeJavaType;
import com.lumen.finance.FinanceConstants;
import com.lumen.finance.engine.BillingExpressionException.FinanceException_UnknownVar;
import org.springframework.stereotype.Component;

import java.math.BigDecimal;
import java.security.MessageDigest;
import java.util.Map;
import java.util.Set;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;

/**
 * Aviator 表达式沙箱引擎 (spec §4).
 *
 * <p><b>沙箱策略 (spec §4.2)</b>:
 * <ul>
 *   <li>白名单输入变量 (8 个: amount / weight / distance / quantity / customer_level / zone / service_type / is_weekend / is_holiday)</li>
 *   <li>禁用 Java 方法调用 ({@code setFunctionMethods(null)} + 禁止自定义 ClassLoader)</li>
 *   <li>禁用属性语法 ({@code DISABLE_PROPERTY_SYNTAX})</li>
 *   <li>100ms 求值超时 (由 {@code AviatorEvaluator} 自身不直接支持, 由调用方在 Future 中控制; spec §4.2 100ms 在沙箱测试中覆盖)</li>
 *   <li>编译缓存 (LRU 1000 上限, 防止内存泄漏)</li>
 * </ul>
 *
 * <p>本类为无状态 Bean, 可被多次复用.
 */
@Component
public class BillingScriptEngine {

    /** 沙箱白名单变量 (spec §4.2). */
    public static final Set<String> WHITELIST_VARIABLES = Set.of(
        "amount", "weight", "distance", "quantity",
        "customer_level", "zone", "service_type",
        "is_weekend", "is_holiday"
    );

    private final AviatorEvaluator evaluator;
    private final Map<String, CompiledExpression> cache = new ConcurrentHashMap<>();

    public BillingScriptEngine() {
        this.evaluator = AviatorEvaluator.newInstance(new AviatorEvaluatorOptions()
            .setOption(Options.ALLOW_NULL_LITERAL, true)
            .setOption(Options.ALWAYS_PARSE_FLOATING_POINT_NUMBER_INTO_DECIMAL, true)
            // 关键安全: 禁用所有 Java 方法反射 + ClassLoader
            .setOption(Options.DISABLE_PROPERTY_SYNTAX, true)
            .setFunctionMethods(null)
            .setLoader(null));
    }

    /**
     * 编译表达式 (含缓存).
     * @throws BillingExpressionException 语法错
     */
    public CompiledExpression compile(String expression) {
        String hash = normalizeHash(expression);
        CompiledExpression cached = cache.get(hash);
        if (cached != null) return cached;

        try {
            Expression compiled = evaluator.compile(expression, true);
            CompiledExpression ce = new CompiledExpression(expression, hash, compiled);
            // LRU: 简化实现 - 如果超上限, 清空一半
            if (cache.size() >= FinanceConstants.EXPRESSION_CACHE_MAX_SIZE) {
                int toRemove = cache.size() / 2;
                cache.keySet().stream().limit(toRemove).forEach(cache::remove);
            }
            cache.put(hash, ce);
            return ce;
        } catch (Exception e) {
            throw new BillingExpressionException(FinanceErrorCode.EXPRESSION_INVALID,
                "Failed to compile expression: " + truncate(expression, 200), e);
        }
    }

    /**
     * 求值 (compile + eval 一次完成).
     * @throws BillingExpressionException 编译失败 / 求值失败
     * @throws BillingExpressionException.FinanceException_UnknownVar 引用未注册变量
     */
    public BigDecimal eval(String expression, Map<String, Object> env) {
        CompiledExpression ce = compile(expression);
        return eval(ce, env);
    }

    /**
     * 用预编译的表达式求值.
     */
    public BigDecimal eval(CompiledExpression ce, Map<String, Object> env) {
        try {
            // 白名单校验: 检测 env 中所有 key 是否都在白名单
            if (env != null) {
                for (String key : env.keySet()) {
                    if (!WHITELIST_VARIABLES.contains(key)) {
                        throw new FinanceException_UnknownVar(key);
                    }
                }
            }
            Object result = ce.getExpression().execute(env);
            if (result instanceof BigDecimal bd) return bd;
            if (result instanceof Number n) return new BigDecimal(n.toString());
            throw new BillingExpressionException(FinanceErrorCode.EXPRESSION_INVALID,
                "Expression did not return a number: " + result);
        } catch (BillingExpressionException ex) {
            throw ex;
        } catch (Exception e) {
            throw new BillingExpressionException(FinanceErrorCode.EXPRESSION_INVALID,
                "Evaluation failed: " + e.getMessage(), e);
        }
    }

    /**
     * 表达式标准化 + SHA-256 hash (spec §4.4 缓存 key).
     */
    public static String normalizeHash(String expression) {
        String normalized = expression.replaceAll("\\s+", "").trim();
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            byte[] hash = md.digest(normalized.getBytes(java.nio.charset.StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder(64);
            for (byte b : hash) sb.append(String.format("%02x", b));
            return sb.toString();
        } catch (Exception e) {
            throw new IllegalStateException("SHA-256 not available", e);
        }
    }

    private static String truncate(String s, int max) {
        if (s == null) return null;
        return s.length() <= max ? s : s.substring(0, max) + "...";
    }
}
```

```java
     * 求值 + 超时控制 (spec §4.2 100ms).
     *
     * <p>用 {@link ExecutorService} + {@link Future#get(long, TimeUnit)} 实现.
     * 到达 timeout 抛 {@link BillingExpressionException}(EXPRESSION_TIMEOUT).
     */
    public BigDecimal eval(String expression, Map<String, Object> env, long timeoutMs) {
        ExecutorService exec = Executors.newSingleThreadExecutor(r -> {
            Thread t = new Thread(r, "billing-eval-" + System.nanoTime());
            t.setDaemon(true);
            return t;
        });
        try {
            Future<BigDecimal> future = exec.submit(() -> eval(expression, env));
            return future.get(timeoutMs, TimeUnit.MILLISECONDS);
        } catch (TimeoutException e) {
            throw new BillingExpressionException(FinanceErrorCode.EXPRESSION_TIMEOUT,
                "Expression evaluation exceeded " + timeoutMs + "ms");
        } catch (ExecutionException e) {
            Throwable cause = e.getCause();
            if (cause instanceof BillingExpressionException bee) throw bee;
            if (cause instanceof RuntimeException re) throw re;
            throw new BillingExpressionException(FinanceErrorCode.EXPRESSION_INVALID,
                "Unexpected error: " + cause.getMessage(), cause);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new BillingExpressionException(FinanceErrorCode.EXPRESSION_TIMEOUT,
                "Interrupted while evaluating");
        } finally {
            exec.shutdownNow();
        }
    }
```

```java
                .eq(BillingRule::getTenantId, tenantId)
                .eq(BillingRule::getId, id));
        if (rule == null) throw new FinanceException(FinanceErrorCode.RULE_NOT_FOUND);
        return toView(rule);
    }

    @Override
    @Transactional
    public BillingRuleDtos.View create(Long tenantId, Long operatorId, BillingRuleDtos.CreateRequest req) {
        // 1. 插 rule 头
        BillingRule rule = new BillingRule();
        rule.setTenantId(tenantId);
        rule.setBusinessType(req.businessType());
        rule.setDirection(req.direction());
        rule.setCounterpartyType(req.counterpartyType());
        rule.setCurrency(req.currency() == null ? FinanceConstants.CURRENCY_CNY : req.currency());
        rule.setPriority(req.priority() == null ? 100 : req.priority());
        rule.setStatus(FinanceConstants.RULE_STATUS_DRAFT);
        rule.setEffectiveFrom(req.effectiveFrom());
        rule.setEffectiveTo(req.effectiveTo());
        rule.setRemark(req.remark());
        rule.setCreatedBy(operatorId);
        try {
            ruleMapper.insert(rule);
        } catch (org.springframework.dao.DuplicateKeyException e) {
            throw new FinanceException(FinanceErrorCode.DUPLICATE_RULE);
        }

        // 2. 插 rule_items
        if (req.items() != null) {
            for (var item : req.items()) {
                BillingRuleItem entity = new BillingRuleItem();
                entity.setRuleId(rule.getId());
                entity.setSequence(item.sequence());
                entity.setItemType(item.itemType());
                entity.setExpression(item.expression());
                entity.setExpressionHash(BillingScriptEngine.normalizeHash(item.expression()));
                entity.setParamJson(item.paramJson());
                entity.setResultField(item.resultField() == null ? "gross_amount" : item.resultField());
                entity.setRemark(item.remark());
                itemMapper.insert(entity);
            }
        }

        // 3. 写 audit
        writeAudit(tenantId, rule.getId(), 0, FinanceConstants.AUDIT_CHANGE_CREATE,
            "{}", operatorId, "Rule created");

        return toView(rule);
    }

    @Override
    @Transactional
    public BillingRuleDtos.View update(Long tenantId, Long id, Long operatorId, BillingRuleDtos.UpdateRequest req) {
        BillingRule rule = ruleMapper.selectOne(
            new LambdaQueryWrapper<BillingRule>()
                .eq(BillingRule::getTenantId, tenantId)
                .eq(BillingRule::getId, id));
        if (rule == null) throw new FinanceException(FinanceErrorCode.RULE_NOT_FOUND);

        // 记录 before 状态用于 audit diff
        String before = toJson(rule);

        if (req.priority() != null) rule.setPriority(req.priority());
        if (req.effectiveTo() != null) rule.setEffectiveTo(req.effectiveTo());
        if (req.remark() != null) rule.setRemark(req.remark());
        rule.setUpdatedBy(operatorId);
        ruleMapper.updateById(rule);

        // 替换 items: 全删全插
        if (req.items() != null) {
            itemMapper.delete(new LambdaQueryWrapper<BillingRuleItem>()
                .eq(BillingRuleItem::getRuleId, id));
            for (var item : req.items()) {
                BillingRuleItem entity = new BillingRuleItem();
                entity.setRuleId(id);
                entity.setSequence(item.sequence());
                entity.setItemType(item.itemType());
                entity.setExpression(item.expression());
                entity.setExpressionHash(BillingScriptEngine.normalizeHash(item.expression()));
                entity.setParamJson(item.paramJson());
                entity.setResultField(item.resultField() == null ? "gross_amount" : item.resultField());
                entity.setRemark(item.remark());
                itemMapper.insert(entity);
            }
        }

        writeAudit(tenantId, id, rule.getVersion(), FinanceConstants.AUDIT_CHANGE_UPDATE,
            "{\"before\":" + before + ",\"after\":" + toJson(rule) + "}", operatorId, "Rule updated");

        return toView(rule);
    }

    @Override
    @Transactional
    public BillingRuleDtos.View activate(Long tenantId, Long id, Long operatorId) {
        BillingRule rule = ruleMapper.selectOne(
            new LambdaQueryWrapper<BillingRule>()
                .eq(BillingRule::getTenantId, tenantId)
                .eq(BillingRule::getId, id));
        if (rule == null) throw new FinanceException(FinanceErrorCode.RULE_NOT_FOUND);
        if (rule.getStatus().equals(FinanceConstants.RULE_STATUS_ACTIVE)) return toView(rule);

        // 编译所有 items, 失败回 INVALID
        List<BillingRuleItem> items = itemMapper.selectList(
            new LambdaQueryWrapper<BillingRuleItem>().eq(BillingRuleItem::getRuleId, id));
        for (BillingRuleItem item : items) {
            try {
                scriptEngine.compile(item.getExpression());
            } catch (BillingExpressionException e) {
                rule.setStatus(FinanceConstants.RULE_STATUS_INVALID);
                ruleMapper.updateById(rule);
                writeAudit(tenantId, id, rule.getVersion(), FinanceConstants.AUDIT_CHANGE_UPDATE,
                    "{\"error\":\"" + e.getMessage() + "\"}", operatorId, "Activation failed");
                throw new FinanceException(FinanceErrorCode.EXPRESSION_INVALID,
                    "Item " + item.getSequence() + " failed: " + e.getMessage());
            }
        }

        rule.setStatus(FinanceConstants.RULE_STATUS_ACTIVE);
        ruleMapper.updateById(rule);
        writeAudit(tenantId, id, rule.getVersion(), FinanceConstants.AUDIT_CHANGE_UPDATE,
            "{\"status\":\"ACTIVE\"}", operatorId, "Rule activated");
        return toView(rule);
    }

    @Override
    @Transactional
    public BillingRuleDtos.View disable(Long tenantId, Long id, Long operatorId) {
        BillingRule rule = ruleMapper.selectOne(
            new LambdaQueryWrapper<BillingRule>()
                .eq(BillingRule::getTenantId, tenantId)
                .eq(BillingRule::getId, id));
        if (rule == null) throw new FinanceException(FinanceErrorCode.RULE_NOT_FOUND);

        rule.setStatus(FinanceConstants.RULE_STATUS_DISABLED);
        ruleMapper.updateById(rule);
        writeAudit(tenantId, id, rule.getVersion(), FinanceConstants.AUDIT_CHANGE_DISABLE,
            "{\"status\":\"DISABLED\"}", operatorId, "Rule disabled");
        return toView(rule);
    }

    private void writeAudit(Long tenantId, Long ruleId, int version, String changeType,
                            String diffJson, Long operatorId, String remark) {
        BillingRuleAudit audit = new BillingRuleAudit();
        audit.setTenantId(tenantId);
        audit.setRuleId(ruleId);
        audit.setRuleVersion(version);
        audit.setChangeType(changeType);
        audit.setDiffJson(diffJson);
        audit.setOperatorId(operatorId);
        audit.setChangedAt(LocalDateTime.now());
        audit.setRemark(remark);
        auditMapper.insert(audit);
    }

    private BillingRuleDtos.View toView(BillingRule rule) {
        List<BillingRuleItem> items = itemMapper.selectList(
            new LambdaQueryWrapper<BillingRuleItem>()
                .eq(BillingRuleItem::getRuleId, rule.getId())
                .orderByAsc(BillingRuleItem::getSequence));
        List<BillingRuleDtos.RuleItemView> itemViews = items.stream()
            .map(i -> new BillingRuleDtos.RuleItemView(
                i.getId(), i.getSequence(), i.getItemType(),
                i.getExpression(), i.getExpressionHash(), i.getResultField(), i.getRemark()))
            .toList();
        return new BillingRuleDtos.View(
            rule.getId(), rule.getBusinessType(), rule.getDirection(),
            rule.getCounterpartyType(), rule.getCurrency(), rule.getPriority(),
            rule.getStatus(), rule.getEffectiveFrom(), rule.getEffectiveTo(),
            rule.getRemark(), rule.getVersion(), itemViews);
    }

    private String toJson(Object o) {
        try { return objectMapper.writeValueAsString(o); }
        catch (JsonProcessingException e) { return "{}"; }
    }
}
```

```java
package com.lumen.finance.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.lumen.finance.FinanceConstants;
import com.lumen.finance.dto.BillingEventDto;
import com.lumen.finance.entity.*;
import com.lumen.finance.exception.FinanceErrorCode;
import com.lumen.finance.exception.FinanceException;
import com.lumen.finance.mapper.*;
import com.lumen.finance.engine.BillingScriptEngine;
import com.lumen.finance.service.BillingEngineService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.DuplicateKeyException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.*;

@Service
@RequiredArgsConstructor
@Slf4j
public class BillingEngineServiceImpl implements BillingEngineService {

    private final BillingRuleMapper ruleMapper;
    private final BillingRuleItemMapper itemMapper;
    private final BillingEventLogMapper eventLogMapper;
    private final BillingAccrualMapper accrualMapper;
    private final BillingScriptEngine scriptEngine;
    private final ObjectMapper objectMapper;

    @Override
    @Transactional
    public BillingAccrual accrue(BillingEventDto e) {
        // 1. 写 event_log (幂等)
        BillingEventLog log = new BillingEventLog();
        log.setTenantId(e.tenantId());
        log.setSourceEventId(e.sourceEventId());
        log.setBusinessType(e.businessType());
        log.setBusinessId(e.businessId());
        log.setReceivedAt(LocalDateTime.now());
        log.setProcessStatus(FinanceConstants.EVT_STATUS_RECEIVED);
        try { eventLogMapper.insert(log); }
        catch (DuplicateKeyException dup) {
            log.info("Duplicate event {} for tenant {}, skip", e.sourceEventId(), e.tenantId());
            return null;
        }

        // 2. 加载匹配 ACTIVE rule (按 priority ASC + effective 范围)
        LocalDate today = LocalDate.now();
        List<BillingRule> candidates = ruleMapper.selectList(
            new LambdaQueryWrapper<BillingRule>()
                .eq(BillingRule::getTenantId, e.tenantId())
                .eq(BillingRule::getBusinessType, e.businessType())
                .eq(BillingRule::getDirection, e.direction())
                .eq(BillingRule::getStatus, FinanceConstants.RULE_STATUS_ACTIVE)
                .le(BillingRule::getEffectiveFrom, today)
                .and(q -> q.isNull(BillingRule::getEffectiveTo).or().ge(BillingRule::getEffectiveTo, today))
                .orderByAsc(BillingRule::getPriority));
        if (candidates.isEmpty()) {
            log.setProcessStatus(FinanceConstants.EVT_STATUS_NO_RULE);
            eventLogMapper.updateById(log);
            log.warn("No ACTIVE rule for tenant={} businessType={} direction={}",
                e.tenantId(), e.businessType(), e.direction());
            return null;
        }
        BillingRule rule = candidates.get(0);

        // 3. 加载 rule_items 按 sequence ASC
        List<BillingRuleItem> items = itemMapper.selectList(
            new LambdaQueryWrapper<BillingRuleItem>()
                .eq(BillingRuleItem::getRuleId, rule.getId())
                .orderByAsc(BillingRuleItem::getSequence));
        if (items.isEmpty()) {
            log.setProcessStatus(FinanceConstants.EVT_STATUS_NO_RULE);
            eventLogMapper.updateById(log);
            return null;
        }

        // 4. 逐个求值, aggregate gross_amount
        BigDecimal gross = BigDecimal.ZERO;
        try {
            for (BillingRuleItem item : items) {
                BigDecimal partial = scriptEngine.eval(item.getExpression(), e.env());
                gross = gross.add(partial);
            }
        } catch (Exception ex) {
            // 求值失败 → 写 accrual (status=OPEN_FAILED) + log
            log.setProcessStatus(FinanceConstants.EVT_STATUS_FAILED);
            log.setErrorMsg(ex.getMessage());
            eventLogMapper.updateById(log);

            BillingAccrual failed = new BillingAccrual();
            failed.setTenantId(e.tenantId());
            failed.setSourceEventId(e.sourceEventId());
            failed.setBusinessType(e.businessType());
            failed.setBusinessId(e.businessId());
            failed.setRuleId(rule.getId());
            failed.setDirection(e.direction());
            failed.setCounterpartyId(e.counterpartyId());
            failed.setGrossAmount(BigDecimal.ZERO);
            failed.setTaxAmount(BigDecimal.ZERO);
            failed.setNetAmount(BigDecimal.ZERO);
            failed.setCurrency(FinanceConstants.CURRENCY_CNY);
            failed.setStatus(FinanceConstants.ACCRUAL_STATUS_OPEN_FAILED);
            failed.setErrorMsg(truncate(ex.getMessage(), 1000));
            try { accrualMapper.insert(failed); } catch (DuplicateKeyException ignored) {}
            throw new FinanceException(FinanceErrorCode.EXPRESSION_INVALID, ex.getMessage());
        }

        // 5. 写 accrual (status=OPEN)
        BillingAccrual accrual = new BillingAccrual();
        accrual.setTenantId(e.tenantId());
        accrual.setSourceEventId(e.sourceEventId());
        accrual.setBusinessType(e.businessType());
        accrual.setBusinessId(e.businessId());
        accrual.setRuleId(rule.getId());
        accrual.setDirection(e.direction());
        accrual.setCounterpartyId(e.counterpartyId());
        accrual.setGrossAmount(gross);
        accrual.setTaxAmount(BigDecimal.ZERO); // 4.3.2 启用
        accrual.setNetAmount(gross);
        accrual.setCurrency(FinanceConstants.CURRENCY_CNY);
        accrual.setStatus(FinanceConstants.ACCRUAL_STATUS_OPEN);
        try {
            accrual.setPayloadJson(toJson(e.env()));
            accrualMapper.insert(accrual);
        } catch (DuplicateKeyException dup) {
            log.info("Duplicate accrual for event {} (tenant={}, businessType={}), skip",
                e.sourceEventId(), e.tenantId(), e.businessType());
            return null;
        }

        // 6. 更新 log
        log.setProcessStatus(FinanceConstants.EVT_STATUS_SUCCESS);
        log.setProcessedAt(LocalDateTime.now());
        eventLogMapper.updateById(log);

        return accrual;
    }

    private String toJson(Object o) {
        try { return objectMapper.writeValueAsString(o); }
        catch (JsonProcessingException e) { return "{}"; }
    }

    private String truncate(String s, int max) {
        if (s == null) return null;
        return s.length() <= max ? s : s.substring(0, max);
    }
}
```

**Step 4: 写 BusinessEventListener (订阅 3 个上游事件)**

```java
package com.lumen.finance.listener;

import com.lumen.common.tenant.TenantContext;
import com.lumen.finance.FinanceConstants;
import com.lumen.finance.dto.BillingEventDto;
import com.lumen.finance.service.BillingEngineService;
import com.lumen.extension.outbox.DomainEvent;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.event.EventListener;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Component;

import java.util.HashMap;
import java.util.Map;

/**
 * 订阅上游业务事件 (spec §6.1).
 *
 * <p>3 个事件: DeliveryOrderStateChangedEvent (COMPLETED) /
 * WmsInboundCompletedEvent / WmsOutboundCompletedEvent.
 * 若 3.0 缺 WMS 事件, 4.3.1 改用轮询 wms_inbound.status=COMPLETED (PR3.5 IT 验证).
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class BusinessEventListener {

    private final BillingEngineService engine;

    @EventListener
    @Async("outboxExecutor")
    public void onEvent(DomainEvent event) {
        try {
            TenantContext.set(event.tenantId());
            BillingEventDto dto = mapToBillingEvent(event);
            if (dto == null) {
                log.debug("Event {} not billable, skip", event.getClass().getSimpleName());
                return;
            }
            engine.accrue(dto);
        } catch (Exception ex) {
            log.error("Failed to process business event {}: {}", event.getClass().getSimpleName(), ex.getMessage(), ex);
        } finally {
            TenantContext.clear();
        }
    }

    private BillingEventDto mapToBillingEvent(DomainEvent event) {
        String className = event.getClass().getSimpleName();
        return switch (className) {
            case "DeliveryOrderStateChangedEvent" -> {
                // 假设 toState=COMPLETED 才计费 (filter)
                Object toState = invokeGetter(event, "getToState");
                if (!"COMPLETED".equals(toState)) yield null;
                Map<String, Object> env = new HashMap<>();
                env.put("weight", invokeGetter(event, "getWeight"));
                env.put("distance", invokeGetter(event, "getDistance"));
                yield new BillingEventDto(
                    event.tenantId(),
                    event.eventId(),
                    FinanceConstants.BIZ_LAST_MILE,
                    ((Number) invokeGetter(event, "getDeliveryOrderId")).longValue(),
                    FinanceConstants.DIR_RECEIVABLE,
                    ((Number) invokeGetter(event, "getCustomerId")).longValue(),
                    FinanceConstants.CP_CUSTOMER,
                    env
                );
            }
            case "WmsInboundCompletedEvent" -> {
                Map<String, Object> env = new HashMap<>();
                env.put("amount", invokeGetter(event, "getServiceFee"));
                yield new BillingEventDto(
                    event.tenantId(), event.eventId(),
                    FinanceConstants.BIZ_WAREHOUSE_IN,
                    ((Number) invokeGetter(event, "getInboundId")).longValue(),
                    FinanceConstants.DIR_PAYABLE,
                    ((Number) invokeGetter(event, "getWarehouseId")).longValue(),
                    FinanceConstants.CP_WAREHOUSE,
                    env
                );
            }
            case "WmsOutboundCompletedEvent" -> {
                Map<String, Object> env = new HashMap<>();
                env.put("amount", invokeGetter(event, "getServiceFee"));
                yield new BillingEventDto(
                    event.tenantId(), event.eventId(),
                    FinanceConstants.BIZ_WAREHOUSE_OUT,
                    ((Number) invokeGetter(event, "getOutboundId")).longValue(),
                    FinanceConstants.DIR_PAYABLE,
                    ((Number) invokeGetter(event, "getWarehouseId")).longValue(),
                    FinanceConstants.CP_WAREHOUSE,
                    env
                );
            }
            default -> null;
        };
    }

    private static Object invokeGetter(Object target, String method) {
        try {
            return target.getClass().getMethod(method).invoke(target);
        } catch (Exception e) { return null; }
    }
}
```

```java
package com.lumen.finance.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.lumen.common.numbering.NumberingService;
import com.lumen.finance.FinanceConstants;
import com.lumen.finance.entity.*;
import com.lumen.finance.exception.FinanceErrorCode;
import com.lumen.finance.exception.FinanceException;
import com.lumen.finance.mapper.*;
import com.lumen.finance.service.AccrualService;
import com.lumen.dispatch.statemachine.StateMachineEngine;  // 复用 4.1
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.List;

@Service
@RequiredArgsConstructor
public class AccrualServiceImpl implements AccrualService {

    private final BillingAccrualMapper accrualMapper;
    private final BillingInvoiceMapper invoiceMapper;
    private final BillingRuleMapper ruleMapper;
    private final StateMachineEngine stateMachine;
    private final NumberingService numberingService;  // lumen-numbering 模块

    @Override
    public List<BillingAccrual> list(Long tenantId, String businessType, String status, int pageNum, int pageSize) {
        LambdaQueryWrapper<BillingAccrual> q = new LambdaQueryWrapper<BillingAccrual>()
            .eq(BillingAccrual::getTenantId, tenantId)
            .orderByDesc(BillingAccrual::getId);
        if (businessType != null) q.eq(BillingAccrual::getBusinessType, businessType);
        if (status != null) q.eq(BillingAccrual::getStatus, status);
        Page<BillingAccrual> page = accrualMapper.selectPage(new Page<>(pageNum, pageSize), q);
        return page.getRecords();
    }

    @Override
    public BillingAccrual get(Long tenantId, Long id) {
        BillingAccrual a = accrualMapper.selectOne(
            new LambdaQueryWrapper<BillingAccrual>()
                .eq(BillingAccrual::getTenantId, tenantId)
                .eq(BillingAccrual::getId, id));
        if (a == null) throw new FinanceException(FinanceErrorCode.ACCRUAL_NOT_FOUND);
        return a;
    }

    @Override
    @Transactional
    public BillingInvoice finalize(Long tenantId, Long accrualId, Long operatorId) {
        BillingAccrual accrual = get(tenantId, accrualId);
        if (!accrual.getStatus().equals(FinanceConstants.ACCRUAL_STATUS_OPEN)) {
            throw new FinanceException(FinanceErrorCode.INVALID_STATE_TRANSITION,
                "Accrual status must be OPEN, got " + accrual.getStatus());
        }

        // 1. 状态机: OPEN → LOCKED
        stateMachine.transition(tenantId, FinanceConstants.SM_BILLING_ACCRUAL, "OPEN", "finalize");
        accrual.setStatus(FinanceConstants.ACCRUAL_STATUS_LOCKED);
        accrual.setLockedAt(LocalDateTime.now());
        accrualMapper.updateById(accrual);

        // 2. 生成 invoice (DRAFT)
        BillingRule rule = ruleMapper.selectById(accrual.getRuleId());
        BillingInvoice inv = new BillingInvoice();
        inv.setInvoiceNo(generateInvoiceNo(tenantId, accrual.getDirection()));
        inv.setTenantId(tenantId);
        inv.setDirection(accrual.getDirection());
        inv.setCounterpartyId(accrual.getCounterpartyId());
        inv.setCounterpartyType(rule.getCounterpartyType());
        inv.setBusinessType(accrual.getBusinessType());
        inv.setSourceAccrualId(accrualId);
        inv.setGrossAmount(accrual.getGrossAmount());
        inv.setTaxAmount(accrual.getTaxAmount());
        inv.setNetAmount(accrual.getNetAmount());
        inv.setCurrency(accrual.getCurrency());
        inv.setStatus(FinanceConstants.INV_STATUS_DRAFT);
        inv.setRuleId(accrual.getRuleId());
        inv.setRuleVersion(rule.getVersion());
        invoiceMapper.insert(inv);

        // 3. 状态机: LOCKED → CONVERTED
        stateMachine.transition(tenantId, FinanceConstants.SM_BILLING_ACCRUAL, "LOCKED", "convert");
        accrual.setStatus(FinanceConstants.ACCRUAL_STATUS_CONVERTED);
        accrualMapper.updateById(accrual);

        return inv;
    }

    @Override
    @Transactional
    public BillingAccrual release(Long tenantId, Long accrualId, String reason, Long operatorId) {
        BillingAccrual accrual = get(tenantId, accrualId);
        if (!accrual.getStatus().equals(FinanceConstants.ACCRUAL_STATUS_OPEN)) {
            throw new FinanceException(FinanceErrorCode.INVALID_STATE_TRANSITION,
                "Only OPEN accrual can be released");
        }
        stateMachine.transition(tenantId, FinanceConstants.SM_BILLING_ACCRUAL, "OPEN", "release");
        accrual.setStatus(FinanceConstants.ACCRUAL_STATUS_RELEASED);
        accrual.setErrorMsg(reason);
        accrualMapper.updateById(accrual);
        return accrual;
    }

    private String generateInvoiceNo(Long tenantId, String direction) {
        String prefix = direction.equals(FinanceConstants.DIR_RECEIVABLE) ? "AR" : "AP";
        for (int attempt = 0; attempt < 3; attempt++) {
            String no = numberingService.next("billing_invoice", prefix, 6);
            if (no != null) return no;
        }
        throw new FinanceException(FinanceErrorCode.INVOICE_NO_CONFLICT);
    }
}
```
