> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** lumen-finance 增加会计期间月结 (MonthlyClosing) + 对账差异自动调账 (Reconciliation) + 3 种 Excel 报表导出 (Report) + Web 报表页. 累计 4.4 末 lumen-finance 单元 92→112 (+20), BE IT 22→26 (+4), Vitest 146→150 (+4), endpoint 34→50 (+16), migrations 30→31.

**Architecture:** V31 migration 加 3 新表 (t_payment_record / t_monthly_closing / t_reconciliation) + 1 状态机 (monthly_closing OPEN→IN_PROGRESS→CLOSED 含 reopen) + 4 RBAC 权限 + 6 sys_role_permission. 3 services (PaymentRecord / MonthlyClosing / Reconciliation) + 1 Report service (Apache POI 即时生成) + 1 ClosingEventListener (异步增量重计快照). 对账自动调账复用 4.3.2 BillingAdjustmentService.create 走 L1/L2 审批.

**Tech Stack:** Spring Boot 3.2.5 + MyBatis-Plus 3.5.7 + Flyway 9.x + Apache POI 5.2 + Apache Commons CSV 1.10 + Lombok + JUnit 5 + Testcontainers. Vite 5 + React 18 + TS 5 + Antd 5 + ProComponents 2.7 + Vitest 1 + RTL + MSW 2.

## 重要基座约定 (实施前必读,沿用 4.3.1/4.3.2)

1. **多租户**: `TenantContext.setTenantId(...)` 在 @BeforeEach 设置; t_payment_record / t_monthly_closing / t_reconciliation 加 MybatisPlusConfig.ignoreTable
2. **StateMachineEngine.assertTransition(smCode, fromState, toState, tenantId)** 已在 lumen-bootstrap,4.4 沿用
3. **MyBatis-Plus BaseMapper**: insert 不会自动 fill createdAt/updatedAt,需用 @TableField(fill=FieldFill.INSERT[_UPDATE]) + MyMetaObjectHandler
4. **金额精度**: BigDecimal + scale 4 (DECIMAL(18,4)), 内部 RoundingMode.HALF_UP; ReportService 输出 scale 2
5. **BillingAdjustmentService.create 复用 4.3.2**: 入参 sourceInvoiceId 必填, amount>0, reason>=10 chars
6. **错误码规范**: FinanceErrorCode 枚举 (按 4.3.1/4.3.2 习惯新增 MONTH_NOT_FOUND 400_020 等)
7. **测试排除**: `BillingRuleMapperTest` 已知 heap-space 问题, 跑 4.4 测试加 `-Dtest='!BillingRuleMapperTest'`
8. **CSV 解析**: lumen-parent 已有 `org.apache.commons:commons-csv:1.10.0` 依赖 (lumen-finance pom 透传), 直接用 CSVFormat.DEFAULT.withFirstRecordAsHeader()

```sql
-- V31 — Iteration 4.4 PR1: 支付记录 + 月结 + 对账 + 报表权限

-- 背景: 4.4 在 4.3.2 (V29/V30) 基础上:
--   - 加 3 张表: t_payment_record (实收/实付) / t_monthly_closing (月结) /
--     t_reconciliation (对账差异)
--   - 加 1 个状态机 monthly_closing (OPEN -> IN_PROGRESS -> CLOSED, 含 reopen)
--   - 加 4 个权限 (id 150..153) + 6 个 sys_role_permission (V30 末 306 -> 312)
--
-- 当前 id 占用 (V30 末态):
--   sys_permission        上限 149
--   sys_role              上限  13
--   sys_role_permission   上限 306
--   t_state_transition    独立 id 列
-- 本次 +0/+0/+6/+3 后, 安全, 无冲突.
-- =============================================================

-- ------------------------------ 1. t_payment_record ------------------------------

CREATE TABLE t_payment_record (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  tenant_id BIGINT NOT NULL,
  period CHAR(7) NOT NULL,
  direction VARCHAR(20) NOT NULL,
  counterparty_id BIGINT NOT NULL,
  counterparty_type VARCHAR(20) NOT NULL,
  currency CHAR(3) NOT NULL,
  amount DECIMAL(18,4) NOT NULL,
  source_ref VARCHAR(100) NULL,
  payment_date DATE NOT NULL,
  source VARCHAR(20) NOT NULL,
  remark VARCHAR(500) NULL,
  created_at DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  created_by BIGINT NOT NULL,
  deleted TINYINT NOT NULL DEFAULT 0,
  UNIQUE KEY uk_payment_tenant_period_cpty (tenant_id, period, direction, counterparty_id, payment_date, deleted)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
;

-- ------------------------------ 2. t_monthly_closing ------------------------------

CREATE TABLE t_monthly_closing (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,

  deleted TINYINT NOT NULL DEFAULT 0,
  UNIQUE KEY uk_closing_tenant_period (tenant_id, period, deleted)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
;

-- ------------------------------ 3. t_reconciliation ------------------------------

CREATE TABLE t_reconciliation (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  tenant_id BIGINT NOT NULL,
  period CHAR(7) NOT NULL,
  direction VARCHAR(20) NOT NULL,

  counterparty_type VARCHAR(20) NOT NULL,
  currency CHAR(3) NOT NULL,
  expected_amount DECIMAL(18,4) NOT NULL,
  actual_amount DECIMAL(18,4) NOT NULL,
  variance_amount DECIMAL(18,4) NOT NULL,
  adjustment_id BIGINT NULL,
  reconciliation_status VARCHAR(20) NOT NULL,
  remark VARCHAR(500) NULL,
  created_at DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  updated_at DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3),
  deleted TINYINT NOT NULL DEFAULT 0,
  UNIQUE KEY uk_recon_tenant_period_cpty (tenant_id, period, direction, counterparty_id, currency, deleted)
```

```java
    public List<BillingPaymentRecord> list(Long tenantId, String period, String direction, int page, int size) {
        LambdaQueryWrapper<BillingPaymentRecord> q = new LambdaQueryWrapper<>();
        if (period != null) q.eq(BillingPaymentRecord::getPeriod, period);
        if (direction != null) q.eq(BillingPaymentRecord::getDirection, direction);
        q.orderByDesc(BillingPaymentRecord::getId);
        return mapper.selectPage(new Page<>(page, size), q).getRecords();
    }

    public BillingPaymentRecord create(Long tenantId, Long userId, CreateRequest req) {
        validatePeriod(req.getPeriod());
        BillingPaymentRecord r = BillingPaymentRecord.builder()
            .tenantId(tenantId).period(req.getPeriod()).direction(req.getDirection())
            .counterpartyId(req.getCounterpartyId()).counterpartyType(req.getCounterpartyType())
            .currency(req.getCurrency()).amount(req.getAmount())
            .sourceRef(req.getSourceRef()).paymentDate(req.getPaymentDate())
            .source(FinanceConstants.PAY_SOURCE_MANUAL).remark(req.getRemark())
            .createdBy(userId)
            .build();
        try {
            mapper.insert(r);
        } catch (DuplicateKeyException e) {
            throw new FinanceException(FinanceErrorCode.PAYMENT_RECORD_DUPLICATE);
        }
        return r;
    }

    @Override
    public ImportResult importCsv(Long tenantId, String period, Long userId, MultipartFile file) {
        validatePeriod(period);
        ImportResult result = new ImportResult();
        List<String> errors = new ArrayList<>();
        int total = 0, success = 0, skipped = 0;
        try (var reader = new InputStreamReader(file.getInputStream(), StandardCharsets.UTF_8);
             var parser = CSVFormat.DEFAULT.withFirstRecordAsHeader().parse(reader)) {
            for (CSVRecord rec : parser) {
                total++;
                try {
                    BillingPaymentRecord r = BillingPaymentRecord.builder()
                        .tenantId(tenantId).period(period)
                        .direction(rec.get("direction"))
                        .counterpartyId(Long.parseLong(rec.get("counterparty_id")))
                        .counterpartyType(rec.get("counterparty_type"))
                        .currency(rec.get("currency"))
                        .amount(new BigDecimal(rec.get("amount")))
                        .paymentDate(LocalDate.parse(rec.get("payment_date")))
                        .sourceRef(rec.isMapped("source_ref") ? rec.get("source_ref") : null)
                        .source(FinanceConstants.PAY_SOURCE_CSV)
                        .remark(rec.isMapped("remark") ? rec.get("remark") : null)
                        .createdBy(userId)
                        .build();
                    mapper.insert(r);
                    success++;
                } catch (DuplicateKeyException e) {
                    skipped++;
                } catch (Exception e) {
                    errors.add("Row " + rec.getRecordNumber() + ": " + e.getMessage());
                }
            }
        } catch (Exception e) {
            throw new FinanceException(FinanceErrorCode.BAD_REQUEST, "CSV parse failed: " + e.getMessage());
        }
        result.setTotalRows(total);
        result.setSuccessCount(success);
        result.setSkippedCount(skipped);
        result.setErrors(errors);
        return result;
    }

    private void validatePeriod(String period) {
        if (period == null || !period.matches(FinanceConstants.PERIOD_PATTERN)) {
            throw new FinanceException(FinanceErrorCode.PAYMENT_RECORD_DUPLICATE);
        }
    }
}
```

```java
package com.lumen.finance.engine;

import com.lumen.finance.FinanceConstants;
import org.springframework.stereotype.Component;
import java.math.BigDecimal;
import java.math.RoundingMode;

@Component
public class SnapshotCalculator {

    public static final int SCALE = 4;
    private static final RoundingMode RM = RoundingMode.HALF_UP;

    /**
     * 把任意币种金额换算到 CNY 基准, 用 fx_rate (从源币到 CNY).
     * fx_rate_at_creation 存的是 "1 源币 = X CNY".
     */
    public BigDecimal toCny(BigDecimal amount, String currency, BigDecimal fxRate) {
        if (FinanceConstants.CURRENCY_CNY.equals(currency)) return amount;
        if (fxRate == null) throw new IllegalArgumentException("fxRate required for non-CNY");
        return amount.multiply(fxRate).setScale(SCALE, RM);
    }

    /**
     * 快照: total_invoice + total_accrual + total_adjustment
     * adjustment: CREDIT 加, DEBIT 减.
     */
    public record SnapshotResult(BigDecimal totalInvoice, BigDecimal totalAccrual, BigDecimal totalAdjustment) {}

    public SnapshotResult empty() {
        return new SnapshotResult(BigDecimal.ZERO, BigDecimal.ZERO, BigDecimal.ZERO);
    }

    public SnapshotResult addInvoice(SnapshotResult acc, BigDecimal amount) {
        return new SnapshotResult(acc.totalInvoice.add(amount), acc.totalAccrual, acc.totalAdjustment);
    }

    public SnapshotResult addAccrual(SnapshotResult acc, BigDecimal amount) {
        return new SnapshotResult(acc.totalInvoice, acc.totalAccrual.add(amount), acc.totalAdjustment);
    }

    public SnapshotResult addAdjustment(SnapshotResult acc, BigDecimal amount, String type) {
        BigDecimal delta = FinanceConstants.ADJ_TYPE_DEBIT.equals(type) ? amount.negate() : amount;
        return new SnapshotResult(acc.totalInvoice, acc.totalAccrual, acc.totalAdjustment.add(delta));
    }
}
```

```java
import java.math.BigDecimal;

import java.util.List;

@Service
@RequiredArgsConstructor
public class MonthlyClosingServiceImpl implements MonthlyClosingService {

    private final BillingMonthlyClosingMapper closingMapper;
    private final BillingInvoiceMapper invoiceMapper;
    private final BillingAccrualMapper accrualMapper;
    private final BillingAdjustmentMapper adjustmentMapper;
    private final SnapshotCalculator snapshotCalculator;
    private final com.lumen.finance.engine.StateMachineEngine stateMachineEngine;
    // 注: StateMachineEngine 在 lumen-bootstrap, lumen-finance 不直接依赖; 4.4 改为通过事件 / 注解方式

        if (c == null) throw new FinanceException(FinanceErrorCode.MONTH_NOT_FOUND);
        return c;
    }

    @Override
    public BillingMonthlyClosing getOrCreate(Long tenantId, String period) {

                .status(FinanceConstants.CLOSING_STATUS_OPEN)

        if (!FinanceConstants.CLOSING_STATUS_OPEN.equals(c.getStatus())) {
            throw new FinanceException(FinanceErrorCode.MONTH_NOT_IN_PROGRESS);
        }
        c.setStatus(FinanceConstants.CLOSING_STATUS_IN_PROGRESS);
        recalculateSnapshotInternal(c);
        closingMapper.updateById(c);
        return c;

            throw new FinanceException(FinanceErrorCode.MONTH_NOT_IN_PROGRESS);

        c.setStatus(FinanceConstants.CLOSING_STATUS_CLOSED);
        c.setClosedAt(LocalDateTime.now());
        recalculateSnapshotInternal(c);
        closingMapper.updateById(c);
        return c;

        BillingMonthlyClosing c = get(tenantId, period);
        c.setStatus(FinanceConstants.CLOSING_STATUS_IN_PROGRESS);
        c.setReopenReason(reason);
        c.setClosedAt(null);
        recalculateSnapshotInternal(c);
        closingMapper.updateById(c);
        return c;
    }

    @Override
    public BillingMonthlyClosing recalculateSnapshot(Long tenantId, String period) {

        recalculateSnapshotInternal(c);
        closingMapper.updateById(c);

        // 简化: 暂只取所有 L2_APPROVED invoice (4.4 暂不按 period 过滤, 真实场景下应加 WHERE DATE_FORMAT)
        // 完整 SQL: SELECT SUM(net_amount * fx_rate_at_creation) ... 后续 PR 优化
        List<BillingInvoice> invoices = invoiceMapper.selectList(
            new LambdaQueryWrapper<BillingInvoice>()
                .eq(BillingInvoice::getStatus, FinanceConstants.INV_STATUS_L2_APPROVED));
        for (BillingInvoice inv : invoices) {
            BigDecimal cny = snapshotCalculator.toCny(
                inv.getNetAmount() == null ? BigDecimal.ZERO : inv.getNetAmount(),
                inv.getCurrency(), inv.getFxRateAtCreation());
            result = snapshotCalculator.addInvoice(result, cny);
        }
        c.setSnapshotTotalInvoice(result.totalInvoice());
        c.setSnapshotTotalAccrual(result.totalAccrual());
        c.setSnapshotTotalAdjustment(result.totalAdjustment());
        c.setSnapshotRecalculatedAt(LocalDateTime.now());
    }
```

```java
package com.lumen.finance.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.lumen.finance.FinanceConstants;
import com.lumen.finance.dto.ReconciliationDtos.*;
import com.lumen.finance.entity.*;
import com.lumen.finance.exception.FinanceErrorCode;
import com.lumen.finance.exception.FinanceException;
import com.lumen.finance.mapper.*;
import com.lumen.finance.service.*;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.math.BigDecimal;
import java.util.*;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class ReconciliationServiceImpl implements ReconciliationService {

    private final BillingReconciliationMapper reconMapper;
    private final BillingInvoiceMapper invoiceMapper;
    private final BillingPaymentRecordMapper paymentMapper;
    private final BillingAdjustmentMapper adjustmentMapper;
    private final BillingAdjustmentService billingAdjustmentService;

    @Override
    public BillingReconciliation get(Long tenantId, Long id) {
        BillingReconciliation r = reconMapper.selectById(id);
        if (r == null) throw new FinanceException(FinanceErrorCode.NOT_FOUND);
        return r;
    }

    @Override
    public List<BillingReconciliation> list(Long tenantId, String period, int page, int size) {
        LambdaQueryWrapper<BillingReconciliation> q = new LambdaQueryWrapper<>();
        if (period != null) q.eq(BillingReconciliation::getPeriod, period);
        q.orderByDesc(BillingReconciliation::getId);
        return reconMapper.selectPage(new Page<>(page, size), q).getRecords();
    }

    @Override
    @Transactional
    public Result runReconciliation(Long tenantId, String period) {
        // 简化: 4.4 暂不考虑 period 过滤 (与 MonthlyClosing 内部一致, 全表 L2_APPROVED)
        List<BillingInvoice> invoices = invoiceMapper.selectList(
            new LambdaQueryWrapper<BillingInvoice>()
                .eq(BillingInvoice::getStatus, FinanceConstants.INV_STATUS_L2_APPROVED));
        List<BillingPaymentRecord> payments = paymentMapper.selectList(
            new LambdaQueryWrapper<BillingPaymentRecord>()
                .eq(BillingPaymentRecord::getPeriod, period));

        // 按 (direction, counterparty_id, currency) 分组
        Map<String, BigDecimal> expected = invoices.stream().collect(Collectors.toMap(
            inv -> key(inv.getDirection(), inv.getCounterpartyId(), inv.getCurrency()),
            inv -> inv.getNetAmount() == null ? BigDecimal.ZERO : inv.getNetAmount(),
            BigDecimal::add));
        Map<String, BigDecimal> actual = payments.stream().collect(Collectors.toMap(
            p -> key(p.getDirection(), p.getCounterpartyId(), p.getCurrency()),
            p -> p.getAmount() == null ? BigDecimal.ZERO : p.getAmount(),
            BigDecimal::add));

        Set<String> allKeys = new HashSet<>();
        allKeys.addAll(expected.keySet());
        allKeys.addAll(actual.keySet());

        List<BillingReconciliation> variances = new ArrayList<>();
        List<String> autoAdjIds = new ArrayList<>();
        int varianceCount = 0;

        for (String k : allKeys) {
            BigDecimal exp = expected.getOrDefault(k, BigDecimal.ZERO);
            BigDecimal act = actual.getOrDefault(k, BigDecimal.ZERO);
            BigDecimal variance = exp.subtract(act);
            if (variance.compareTo(BigDecimal.ZERO) == 0) continue;

            // 幂等: 已存在 ADJUSTMENT_CREATED 跳过
            String[] parts = k.split("\\|");
            String direction = parts[0];
            Long counterpartyId = Long.parseLong(parts[1]);
            String currency = parts[2];
            BillingReconciliation existing = reconMapper.selectOne(
                new LambdaQueryWrapper<BillingReconciliation>()
                    .eq(BillingReconciliation::getPeriod, period)
                    .eq(BillingReconciliation::getDirection, direction)
                    .eq(BillingReconciliation::getCounterpartyId, counterpartyId)
                    .eq(BillingReconciliation::getCurrency, currency));
            if (existing != null && FinanceConstants.RECON_STATUS_ADJUSTMENT_CREATED
                    .equals(existing.getReconciliationStatus())) {
                continue;
            }

            // 自动调账: variance>0 RECEIVABLE -> DEBIT, variance<0 RECEIVABLE -> CREDIT
            String adjType;
            if (FinanceConstants.DIR_RECEIVABLE.equals(direction)) {
                adjType = variance.compareTo(BigDecimal.ZERO) > 0
                    ? FinanceConstants.ADJ_TYPE_DEBIT
                    : FinanceConstants.ADJ_TYPE_CREDIT;
            } else {
                adjType = variance.compareTo(BigDecimal.ZERO) > 0
                    ? FinanceConstants.ADJ_TYPE_CREDIT
                    : FinanceConstants.ADJ_TYPE_DEBIT;
            }
            BillingAdjustmentService.CreateRequest adjReq = new BillingAdjustmentService.CreateRequest();
            adjReq.setAdjustmentType(adjType);
            adjReq.setAmount(variance.abs());
            adjReq.setCurrency(currency);
            adjReq.setSourceInvoiceId(null);  // 4.4 暂不强求 source_invoice
            adjReq.setReason("对账差异自动生成: 期间 " + period + ", 差异 " + variance);

            BillingAdjustment adjustment = billingAdjustmentService.create(tenantId, adjReq);
            autoAdjIds.add(String.valueOf(adjustment.getId()));

            // 写 t_reconciliation
            BillingReconciliation recon = BillingReconciliation.builder()
                .tenantId(tenantId).period(period).direction(direction)
                .counterpartyId(counterpartyId).counterpartyType(extractCounterpartyType(payments, counterpartyId))
                .currency(currency)
                .expectedAmount(exp).actualAmount(act).varianceAmount(variance)
                .adjustmentId(adjustment.getId())
                .reconciliationStatus(FinanceConstants.RECON_STATUS_ADJUSTMENT_CREATED)
                .build();
            reconMapper.insert(recon);
            variances.add(recon);
            varianceCount++;
        }

        Result r = new Result();
        r.setPeriod(period);
        r.setTotalChecked(allKeys.size());
        r.setVarianceCount(varianceCount);
        r.setVariances(variances);
        r.setAutoAdjustmentIds(autoAdjIds);
        return r;
    }

    @Override
    public BillingReconciliation waive(Long tenantId, Long id, String reason) {
        if (reason == null || reason.length() < FinanceConstants.ADJUSTMENT_REASON_MIN_LEN) {
            throw new FinanceException(FinanceErrorCode.BAD_REQUEST,
                "waive reason 至少 " + FinanceConstants.ADJUSTMENT_REASON_MIN_LEN + " 字符");
        }
        BillingReconciliation r = get(tenantId, id);
        r.setReconciliationStatus(FinanceConstants.RECON_STATUS_WAIVED);
        r.setRemark(reason);
        reconMapper.updateById(r);
        return r;
    }

    private String key(String direction, Long counterpartyId, String currency) {
        return direction + "|" + counterpartyId + "|" + currency;
    }

    private String extractCounterpartyType(List<BillingPaymentRecord> payments, Long counterpartyId) {
        return payments.stream()
            .filter(p -> p.getCounterpartyId().equals(counterpartyId))
            .map(BillingPaymentRecord::getCounterpartyType)
            .findFirst().orElse("CUSTOMER");
    }
}
```

```java
package com.lumen.finance.service.impl;

import com.lumen.finance.FinanceConstants;
import com.lumen.finance.dto.ReportDtos.ExportRequest;
import com.lumen.finance.dto.ReportDtos.ReportType;
import com.lumen.finance.entity.*;

import com.lumen.finance.service.ReportService;
import lombok.RequiredArgsConstructor;
import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.streaming.SXSSFWorkbook;
import org.springframework.stereotype.Service;
import java.io.ByteArrayOutputStream;
import java.math.BigDecimal;
import java.util.List;

@Service
@RequiredArgsConstructor
public class ReportServiceImpl implements ReportService {

    private final BillingMonthlyClosingMapper closingMapper;
    private final BillingInvoiceMapper invoiceMapper;
    private final BillingAdjustmentMapper adjustmentMapper;
    private final BillingReconciliationMapper reconMapper;

    @Override
    public byte[] export(ExportRequest req, Long tenantId) {
        try (SXSSFWorkbook wb = new SXSSFWorkbook(100)) {
            Sheet sheet = wb.createSheet(req.getType().name());
            switch (req.getType()) {
                case SUMMARY -> writeSummary(wb, sheet, req, tenantId);
                case DETAIL  -> writeDetail(wb, sheet, req, tenantId);
                case VARIANCE -> writeVariance(wb, sheet, req, tenantId);
            }
            ByteArrayOutputStream baos = new ByteArrayOutputStream();
            wb.write(baos);
            return baos.toByteArray();
        } catch (Exception e) {
            throw new RuntimeException("Report generation failed: " + e.getMessage(), e);
        }
    }

    private void writeSummary(SXSSFWorkbook wb, Sheet sheet, ExportRequest req, Long tenantId) {
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("Item");
        header.createCell(1).setCellValue("Currency");
        header.createCell(2).setCellValue("Amount (CNY)");
        header.createCell(3).setCellValue("Period");

        BillingMonthlyClosing closing = closingMapper.selectOne(
            new com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper<BillingMonthlyClosing>()
                .eq(BillingMonthlyClosing::getPeriod, req.getPeriod()));
        if (closing == null) {
            Row noData = sheet.createRow(1);
            noData.createCell(0).setCellValue("No Data");
            return;
        }
        int r = 1;
        addDataRow(sheet, r++, "Invoice",      closing.getSnapshotCurrency(),
            closing.getSnapshotTotalInvoice(), req.getPeriod());
        addDataRow(sheet, r++, "Accrual",      closing.getSnapshotCurrency(),
            closing.getSnapshotTotalAccrual(), req.getPeriod());
        addDataRow(sheet, r++, "Adjustment",   closing.getSnapshotCurrency(),
            closing.getSnapshotTotalAdjustment(), req.getPeriod());
    }

    private void writeDetail(SXSSFWorkbook wb, Sheet sheet, ExportRequest req, Long tenantId) {
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("Invoice No");
        header.createCell(1).setCellValue("Status");
        header.createCell(2).setCellValue("Counterparty");
        header.createCell(3).setCellValue("Currency");
        header.createCell(4).setCellValue("Net Amount");
        header.createCell(5).setCellValue("FX Rate");
        header.createCell(6).setCellValue("Adjustment Credit");
        header.createCell(7).setCellValue("Adjustment Debit");

        List<BillingInvoice> invoices = invoiceMapper.selectList(
            new com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper<BillingInvoice>()
                .eq(BillingInvoice::getStatus, FinanceConstants.INV_STATUS_L2_APPROVED));
        int r = 1;
        for (BillingInvoice inv : invoices) {
            Row row = sheet.createRow(r++);
            row.createCell(0).setCellValue(inv.getInvoiceNo());
            row.createCell(1).setCellValue(inv.getStatus());
            row.createCell(2).setCellValue(String.valueOf(inv.getCounterpartyId()));
            row.createCell(3).setCellValue(inv.getCurrency());
            row.createCell(4).setCellValue(inv.getNetAmount() == null ? 0 : inv.getNetAmount().doubleValue());
            row.createCell(5).setCellValue(inv.getFxRateAtCreation() == null ? 1.0
                : inv.getFxRateAtCreation().doubleValue());
        }
    }

    private void writeVariance(SXSSFWorkbook wb, Sheet sheet, ExportRequest req, Long tenantId) {
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("Period");
        header.createCell(1).setCellValue("Direction");
        header.createCell(2).setCellValue("Counterparty");
        header.createCell(3).setCellValue("Currency");
        header.createCell(4).setCellValue("Expected");
        header.createCell(5).setCellValue("Actual");
        header.createCell(6).setCellValue("Variance");
        header.createCell(7).setCellValue("Status");
        header.createCell(8).setCellValue("Adjustment ID");

        List<BillingReconciliation> recs = reconMapper.selectList(
            new com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper<BillingReconciliation>()
                .eq(BillingReconciliation::getPeriod, req.getPeriod()));
        int r = 1;
        for (BillingReconciliation rec : recs) {
            Row row = sheet.createRow(r++);
            row.createCell(0).setCellValue(rec.getPeriod());
            row.createCell(1).setCellValue(rec.getDirection());
            row.createCell(2).setCellValue(String.valueOf(rec.getCounterpartyId()));
            row.createCell(3).setCellValue(rec.getCurrency());
            row.createCell(4).setCellValue(rec.getExpectedAmount().doubleValue());
            row.createCell(5).setCellValue(rec.getActualAmount().doubleValue());
            row.createCell(6).setCellValue(rec.getVarianceAmount().doubleValue());
            row.createCell(7).setCellValue(rec.getReconciliationStatus());
            row.createCell(8).setCellValue(rec.getAdjustmentId() == null ? "" : String.valueOf(rec.getAdjustmentId()));
        }
    }

    private void addDataRow(Sheet sheet, int row, String item, String currency, BigDecimal amount, String period) {
        Row r = sheet.createRow(row);
        r.createCell(0).setCellValue(item);
        r.createCell(1).setCellValue(currency);
        r.createCell(2).setCellValue(amount == null ? 0 : amount.doubleValue());
        r.createCell(3).setCellValue(period);
    }
}
```
