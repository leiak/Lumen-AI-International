**Step 2: Write CurrencyConverter**

```java
// lumen-parent/lumen-finance/src/main/java/com/lumen/finance/engine/CurrencyConverter.java
package com.lumen.finance.engine;

import org.springframework.stereotype.Component;
import java.math.BigDecimal;
import java.math.RoundingMode;

/**
 * 4.3.2 multi-currency + tax 数学统一计算 (Iteration 4.3.2 PR2-T2.1).
 *
 * <p>含税价/不含税价 风格 (CNY 财务惯例): amount 含税, tax = amount * rate / (1+rate), net = amount - tax.</p>
 *
 * <p>RoundingMode.HALF_UP, 4 位小数 (与 t_billing_invoice / t_billing_adjustment 列精度一致).</p>
 */
@Component
public class CurrencyConverter {

    public static final int SCALE = 4;

    /**
     * 含税价 → 拆税. tax = amount * rate / (1+rate).
     */
    public static BigDecimal calculateTax(BigDecimal grossAmount, BigDecimal taxRate) {
        if (grossAmount == null || taxRate == null) {
            throw new IllegalArgumentException("amount and taxRate required");
        }
        if (taxRate.signum() == 0) return BigDecimal.ZERO.setScale(SCALE, RoundingMode.HALF_UP);
        BigDecimal denominator = BigDecimal.ONE.add(taxRate);
        return grossAmount.multiply(taxRate).divide(denominator, SCALE, RoundingMode.HALF_UP);
    }

    /**
     * 拆含税价为 [tax, net].
     */
    public static BigDecimal[] splitGross(BigDecimal grossAmount, BigDecimal taxRate) {
        BigDecimal tax = calculateTax(grossAmount, taxRate);
        BigDecimal net = grossAmount.subtract(tax).setScale(SCALE, RoundingMode.HALF_UP);
        return new BigDecimal[]{tax, net};
    }

    /**
     * 跨币种转换. fxRate = 1 from = X to.
     */
    public static BigDecimal convert(BigDecimal amount, String from, String to, BigDecimal fxRate) {
        if (amount == null || fxRate == null) {
            throw new IllegalArgumentException("amount and fxRate required");
        }
        if (from.equals(to)) return amount.setScale(SCALE, RoundingMode.HALF_UP);
        return amount.multiply(fxRate).setScale(SCALE, RoundingMode.HALF_UP);
    }
}
```

```java
package com.lumen.finance.rates;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.lumen.finance.exception.FinanceException;
import com.lumen.finance.exception.FinanceErrorCode;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestTemplate;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

@Component
@RequiredArgsConstructor
@Slf4j
public class ExchangeRateHostProvider implements RateProvider {
    private static final String BASE_URL = "https://api.exchangerate.host/latest";
    private final RestTemplate restTemplate;
    private final ObjectMapper objectMapper = new ObjectMapper();
    
    @Override
    public RateQuote fetch(String from, String to, LocalDate asOf) {
        String url = BASE_URL + "?base=" + from + "&symbols=" + to;
        try {
            String response = restTemplate.getForObject(url, String.class);
            JsonNode root = objectMapper.readTree(response);
            if (!root.path("success").asBoolean()) {
                throw new FinanceException(FinanceErrorCode.RATE_SYNC_FAILED,
                    "exchangerate.host returned success=false: " + response);
            }
            BigDecimal rate = new BigDecimal(root.path("rates").path(to).asText());
            return new RateQuote(rate, asOf, "EXCHANGERATE_HOST");
        } catch (Exception e) {
            log.error("Failed to fetch rate from exchangerate.host: {}", e.getMessage());
            throw new FinanceException(FinanceErrorCode.RATE_SYNC_FAILED,
                "Failed to fetch rate: " + e.getMessage(), e);
        }
    }
}
```

```java
import org.springframework.stereotype.Service;
```

```java
@RequiredArgsConstructor
public class ExchangeRateService {
    private final BillingExchangeRateMapper exchangeRateMapper;
    
    /**
     * 取 today 或之前最近一条 (any source). 若完全无数据, 抛 RATE_UNAVAILABLE.
     */
    public BigDecimal getRate(String from, String to, LocalDate asOf) {
        LambdaQueryWrapper<BillingExchangeRate> q = new LambdaQueryWrapper<>();
        q.eq(BillingExchangeRate::getFromCurrency, from)
         .eq(BillingExchangeRate::getToCurrency, to)
         .le(BillingExchangeRate::getRateDate, asOf)
         .orderByDesc(BillingExchangeRate::getRateDate)
         .last("LIMIT 1");
        BillingExchangeRate record = exchangeRateMapper.selectOne(q);
        if (record == null) {
            throw new FinanceException(FinanceErrorCode.RATE_UNAVAILABLE,
                "No exchange rate found for " + from + "->" + to + " as of " + asOf);
        }
        return record.getRate();
    }
    
    public Optional<BigDecimal> getRateOnDate(String from, String to, LocalDate date) {
        LambdaQueryWrapper<BillingExchangeRate> q = new LambdaQueryWrapper<>();
        q.eq(BillingExchangeRate::getFromCurrency, from)
         .eq(BillingExchangeRate::getToCurrency, to)
         .eq(BillingExchangeRate::getRateDate, date);
        BillingExchangeRate record = exchangeRateMapper.selectOne(q);
        return Optional.ofNullable(record).map(BillingExchangeRate::getRate);
    }
    
    public List<BillingExchangeRate> list(String from, String to, LocalDate fromDate, LocalDate toDate, int page, int size) {
        LambdaQueryWrapper<BillingExchangeRate> q = new LambdaQueryWrapper<>();
        if (from != null) q.eq(BillingExchangeRate::getFromCurrency, from);
        if (to != null) q.eq(BillingExchangeRate::getToCurrency, to);
        if (fromDate != null) q.ge(BillingExchangeRate::getRateDate, fromDate);
        if (toDate != null) q.le(BillingExchangeRate::getRateDate, toDate);
        q.orderByDesc(BillingExchangeRate::getRateDate);
        return exchangeRateMapper.selectList(q);
    }
    
    public void saveManual(String from, String to, BigDecimal rate, LocalDate rateDate, Long operatorId) {
        BillingExchangeRate record = BillingExchangeRate.builder()
            .fromCurrency(from)
            .toCurrency(to)
            .rate(rate)
            .rateDate(rateDate)
            .source("MANUAL")
            .createdAt(java.time.LocalDateTime.now())
            .build();
        exchangeRateMapper.insert(record);
    }
}
```
