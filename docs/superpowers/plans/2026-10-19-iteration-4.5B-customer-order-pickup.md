## 重要执行注意事项 (每次 PR 开头先读)

### 既有约定
1. **包命名**: `com.lumen.<module>.*` (lumen-customer 用 `com.lumen.customer.*`)
2. **MyBatis-Plus**: 实体 `@TableName("t_xxx")`, mapper 继承 `BaseMapper<T>`, 多租户用 `TenantContext.getTenantId()` (ThreadLocal)
3. **RBAC**: controller 上 `@PreAuthorize("hasAuthority('xxx:yyy:zzz')")`; 错误码用 `RbacException` / `<Module>Exception`
4. **DTO 校验**: jakarta.validation (`@NotBlank` / `@Size` / `@Email` 等)
5. **统一响应**: `R<T>` (lumen-common 既有) — 成功 `R.ok(data)`, 失败 `R.fail(code, msg)`
6. **测试**: JUnit 5 (`@Test` / `@BeforeEach` / Mockito `@Mock` / `@InjectMocks`), IT 用 `@SpringBootTest` + Testcontainers MySQL
7. **Flyway migration**: 编号连续 (V36), 必须 `ON DUPLICATE KEY UPDATE` 幂等
8. **commit message**: 沿用 4.x 约定 `feat(4.5B.x): ...` / `test(4.5B.x): ...` / `docs(4.5B.x): ...` / `fix(4.5B.x): ...`

### 客户范围执行铁律 (任何 PR 涉及 portal 端点必须遵循)
1. **任何 portal controller 方法** 第一行必须 `CustomerContext.requireCustomerId()` (拦截器已注入, 但显式断言防 NPE)
2. **任何 service 重载** 第一行必须用 `customerId` 做范围过滤, 不允许"先查后过滤"
3. **任何 customer 写操作** 必须 audit log, 带 `audience=portal` + `customer_id`
4. **任何 customer 查询结果** 必须 DTO 转换, 不返 `tenant_id` / `created_by` / `updated_by` 字段 (防止 staff 信息泄露)

### 每次 commit 前自检
- `mvn -pl <module> -am test` (本地模块单测)
- `mvn -pl <module> -am verify -DskipITs=false` (含 IT)
- 若改了 `lumen-common`: 跑 `mvn -pl lumen-common -am test` 后再跑 `mvn -pl lumen-customer -am test` (下游受影响)
- commit message 严格按 `<type>(<scope>): <subject>` 格式

### 关键表名差异 (spec vs DB 实际)
- spec §5.4 写 `sys_state_transition.sm_key` / `action` / `allowed_role`, 但 DB 实际列名 (V8 + V20) 是 `state_machine_code` / `transition_name` / `guard_expression`. 4.5B 沿用 DB 实际列名, 不改 schema.
- V36 状态机扩展 INSERT 写 `state_machine_code='DISPATCH_ORDER'`, `transition_name='cancel'`, `guard_expression='role:customer'`, `sort_order=35` (staff cancel 是 30/31).

```sql
-- =============================================================
-- V36: t_customer_rate_override 表 (Iteration 4.5B.1)
-- =============================================================
-- 背景:
--   4.5B 引入合同价简化版: 客户 (B2B) 与物流商签订固定价 (按 business_type 维度),
--   计费时 BillingEngineService.estimate 优先匹配 override, 命中直接返 base_amount,
--   跳过 BillingRule Aviator 求值 (性能 < 200ms 兜底). 简化合同价谈判/议价 (4.5B 不做).

CREATE TABLE t_customer_rate_override (
    id              BIGINT       NOT NULL AUTO_INCREMENT,
    tenant_id       BIGINT       NOT NULL,
    customer_id     BIGINT       NOT NULL,
    business_type   VARCHAR(32)  NOT NULL COMMENT 'LAST_MILE / WAREHOUSING / etc, 同 billing_rule.business_type',
    base_amount     DECIMAL(18,4) NOT NULL COMMENT '固定基础价 (含税, 币种同 tenant.default_currency)',
    currency        VARCHAR(8)   NOT NULL COMMENT 'ISO 4217, 默认 CNY',
    effective_from  DATE         NOT NULL,
    effective_to    DATE         NULL    COMMENT 'NULL = 长期有效',
    remark          VARCHAR(255) NULL,
    deleted         TINYINT      NOT NULL DEFAULT 0,
    created_at      DATETIME     NOT NULL,
    updated_at      DATETIME     NOT NULL,
    version         INT          NOT NULL DEFAULT 0,
    PRIMARY KEY (id),
    UNIQUE KEY uk_tenant_customer_biz (tenant_id, customer_id, business_type, effective_from),
    KEY idx_override_active (tenant_id, customer_id, business_type, effective_from, effective_to, deleted)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
    COMMENT='商家合同价 override (优先于 BillingRule 跑 Aviator 求值)';
```

```java
@Override
public EstimateResult estimate(Long tenantId, Long customerId, String businessType, Map<String, Object> env) {
    // 1. override 优先
    Optional<CustomerRateOverride> overrideOpt = overrideService.findActive(
        tenantId, customerId, businessType, LocalDate.now());
    if (overrideOpt.isPresent()) {
        CustomerRateOverride o = overrideOpt.get();
        String breakdown = String.format("合同价 %s %s (override #%d)",
            o.getCurrency(), o.getBaseAmount().toPlainString(), o.getId());
        return new EstimateResult(o.getBaseAmount(), o.getCurrency(),
            EstimateResult.SOURCE_RATE_OVERRIDE, o.getId(), breakdown);
    }

    // 2. 匹配 ACTIVE rule (复用 accrue 的查询条件, 但只取 rule 不写库)
    LambdaQueryWrapper<BillingRule> ruleQ = new LambdaQueryWrapper<BillingRule>()
        .eq(BillingRule::getTenantId, tenantId)
        .eq(BillingRule::getBusinessType, businessType)
        .eq(BillingRule::getStatus, FinanceConstants.RULE_STATUS_ACTIVE)
        .le(BillingRule::getEffectiveFrom, LocalDate.now())
        .and(w -> w.isNull(BillingRule::getEffectiveTo)
            .or().ge(BillingRule::getEffectiveTo, LocalDate.now()))
        .orderByAsc(BillingRule::getPriority)
        .orderByAsc(BillingRule::getId)
        .last("LIMIT 1");

    BillingRule rule = ruleMapper.selectOne(ruleQ);
    if (rule == null) {
        return EstimateResult.none();
    }

    List<BillingRuleItem> items = itemMapper.selectList(
        new LambdaQueryWrapper<BillingRuleItem>()
            .eq(BillingRuleItem::getRuleId, rule.getId())
            .orderByAsc(BillingRuleItem::getSequence));

    if (items.isEmpty()) {
        return EstimateResult.none();
    }

    // 3. 逐条求值
    BigDecimal gross = BigDecimal.ZERO;
    StringBuilder breakdown = new StringBuilder();
    for (BillingRuleItem item : items) {
        BigDecimal partial = scriptEngine.eval(item.getExpression(), env);
        partial = partial == null ? BigDecimal.ZERO : partial;
        gross = gross.add(partial);
        if (breakdown.length() > 0) breakdown.append(" + ");
        breakdown.append(item.getExpression()).append(" ").append(partial.toPlainString());
    }
    String currency = rule.getCurrency() == null ? FinanceConstants.CURRENCY_CNY : rule.getCurrency();
    return new EstimateResult(gross, currency, EstimateResult.SOURCE_RULE_AUDIT,
        rule.getId(), breakdown + " = " + gross.toPlainString());
}
```

```java
@Override
public void estimateAndAccrue(BillingEventDto event) {
    try {
        // 1. 同步估价
        EstimateResult result = estimate(event.tenantId(), event.counterpartyId(),
            event.businessType(), event.env());

        // 2. 构造 BillingEventDto 给 accrue (复用)
        // 注: sourceEventId = doId 已经在 event 入参里
        if (EstimateResult.SOURCE_NONE.equals(result.source())) {
            log.info("estimateAndAccrue: NO_RULE (tenant={}, sourceEventId={}), skip accrual",
                event.tenantId(), event.sourceEventId());
            return;
        }

        // 3. 调 accrue 写 OPEN accrual
        //   accrue 内部已经处理 override / rule / 求值 / accrual 落地
        BillingAccrual accrual = accrue(event);
        if (accrual == null) {
            log.info("estimateAndAccrue: accrue returned null (duplicate or NO_RULE) for sourceEventId={}",
                event.sourceEventId());
        }
    } catch (FinanceException ex) {
        // 失败路径: 在 REQUIRES_NEW 事务里写 OPEN_FAILED accrual
        log.warn("estimateAndAccrue failed for sourceEventId={}: {}",
            event.sourceEventId(), ex.getMessage());
        requiresNewTx.executeWithoutResult(status ->
            writeEstimateAndAccrueFailure(event, ex.getMessage())
        );
        // 不抛 — DO 创建是关键路径, finance 失败由后台 job 重试
    } catch (Exception ex) {
        log.warn("estimateAndAccrue unexpected error for sourceEventId={}",
            event.sourceEventId(), ex);
        requiresNewTx.executeWithoutResult(status ->
            writeEstimateAndAccrueFailure(event, ex.getMessage())
        );
    }
}

private void writeEstimateAndAccrueFailure(BillingEventDto event, String err) {
    BillingAccrual accrual = new BillingAccrual();
    accrual.setTenantId(event.tenantId());
    accrual.setSourceEventId(event.sourceEventId());
    accrual.setBusinessType(event.businessType());
    accrual.setBusinessId(event.businessId());
    accrual.setDirection(event.direction());
    accrual.setCounterpartyId(event.counterpartyId());
    accrual.setGrossAmount(BigDecimal.ZERO);
    accrual.setTaxAmount(BigDecimal.ZERO);
    accrual.setNetAmount(BigDecimal.ZERO);
    accrual.setCurrency(FinanceConstants.CURRENCY_CNY);
    accrual.setStatus(FinanceConstants.ACCRUAL_STATUS_OPEN_FAILED);
    accrual.setErrorMsg(truncate("estimateAndAccrue: " + err, 500));
    try {
        accrualMapper.insert(accrual);
    } catch (DuplicateKeyException dup) {
```

```java
                                          PortalOrderRequest req) {

    validatePickupWindow(req);

    // 2. 构造 DO
    WmsDeliveryOrder order = new WmsDeliveryOrder();
    order.setTenantId(tenantId);
    order.setCustomerId(customerId);
    order.setNo("PO-" + UUID.randomUUID().toString().substring(0, 8).toUpperCase());
    order.setSourceType("PORTAL");  // 4.5B 新枚举值 (Task 2.5)
    order.setStatus("PENDING_DISPATCH");
    order.setSignRequired(1);
    order.setWarehouseId(req.warehouseId());
    // 注: 既有字段名视 4.1.x 末态定. shipperAddress/consigneeAddress 可能叫 originAddress/destAddress.
    // 4.5B 落地按 CreateDeliveryOrderRequest 既有 setter 拼.
    order.setSourceNo(req.shipperAddress());  // 占位: 实际 4.5B 看 CreateDeliveryOrderRequest 字段
    order.setTotalWeightKg(req.totalWeightKg());
    order.setTotalVolumeM3(req.totalVolumeM3());
    order.setStopCount(req.stopCount());
    order.setPlannedPickupWindowStart(req.plannedPickupWindowStart());
    // pickup window end: null → start + 4h
    order.setPlannedPickupWindowEnd(req.plannedPickupWindowEnd() != null
        ? req.plannedPickupWindowEnd()
        : req.plannedPickupWindowStart().plusHours(4));
    order.setRemark(req.remark());

    // 3. insert
    wmsDeliveryOrderMapper.insert(order);

    // 4. 同步 estimateAndAccrue (失败不阻塞 DO)
    try {
        BillingEventDto event = new BillingEventDto(
            tenantId,
            order.getId(),  // sourceEventId = doId
            "LAST_MILE",
            order.getId(),  // businessId = doId
            "RECEIVABLE",
            customerId,     // counterpartyId = customerId
            Map.of(
                "weight", req.totalWeightKg(),
                "stopCount", req.stopCount(),
                "distance", BigDecimal.ZERO,  // 4.5B 暂不接地图 API
                "totalWeightKg", req.totalWeightKg()
            )
        );
        billingEngineService.estimateAndAccrue(event);
    } catch (Exception ex) {
        log.warn("createForCustomer: estimateAndAccrue failed for doId={}, reason={}",
            order.getId(), ex.getMessage());
        // 不抛 — DO 创建是关键路径
    }

    // 5. 发事件 (via eventBus, 走 outbox)
    eventBus.publish(new DeliveryOrderStateChangedEvent(
        order, null, "PENDING_DISPATCH", "create"));

    return order;
}

private void validatePickupWindow(PortalOrderRequest req) {
    LocalDateTime now = LocalDateTime.now();
    LocalDateTime start = req.plannedPickupWindowStart();
    LocalDateTime end = req.plannedPickupWindowEnd() != null
        ? req.plannedPickupWindowEnd()
        : start.plusHours(4);

    if (start.isBefore(now.plusHours(2)) || start.isAfter(now.plusDays(30))) {
        throw new com.lumen.customer.exception.CustomerException(400_002,
            "取件时间窗口需 >= 2 小时后, <= 30 天内");
    }
    if (!end.isAfter(start) || end.isAfter(start.plusHours(8))) {
        throw new com.lumen.customer.exception.CustomerException(400_001,
            "取件时间窗口终点需 > 起点, <= 起点 + 8 小时");
    }
    if (!end.toLocalDate().equals(start.toLocalDate())) {
        throw new com.lumen.customer.exception.CustomerException(400_001,
            "取件时间窗口需同日");
    }
}
```

```java
@Override
@Transactional
public WmsDeliveryOrder cancelForCustomer(Long tenantId, Long customerId, Long doId, String reason) {
    WmsDeliveryOrder order = wmsDeliveryOrderMapper.selectById(doId);
    // ownership: 跨租户 OR 跨客户 → 404 (统一不暴露存在性)
    if (order == null
        || !Objects.equals(order.getTenantId(), tenantId)
        || !Objects.equals(order.getCustomerId(), customerId)) {
        throw new DeliveryOrderNotFoundException(doId);
    }
    // 状态 guard: 仅 PENDING_DISPATCH 可取消
    if (!"PENDING_DISPATCH".equals(order.getStatus())) {
        throw new com.lumen.customer.exception.CustomerException(
            com.lumen.customer.error.PortalErrorCode.ORDER_NOT_CANCELLABLE.getCode(),
            "订单当前状态不可取消 (仅 PENDING_DISPATCH 可取消)");
    }
    // 改状态 + remark 后缀
    order.setStatus("CANCELLED");
    String oldRemark = order.getRemark() == null ? "" : order.getRemark();
    order.setRemark(oldRemark + "\n[cancel by customer " + customerId + ": " + reason + "]");
    wmsDeliveryOrderMapper.updateById(order);
    // 发事件
    eventBus.publish(new DeliveryOrderStateChangedEvent(
        order, "PENDING_DISPATCH", "CANCELLED", "cancel"));
    return order;
}
```
