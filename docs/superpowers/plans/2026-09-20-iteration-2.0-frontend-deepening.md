- [x] **Step 2: 写 `AuthProvider.tsx`**

```tsx
import { createContext, useCallback, useEffect, useMemo, useState } from 'react';
import { request } from '@/services/request';
import type { AuthContextValue, CurrentUser } from './types';
import { clearAuthTokens, getAccessToken, getRefreshToken, setAuthTokens } from './tokenStorage';

export const AuthContext = createContext<AuthContextValue | null>(null);

const TOKEN_KEY_ACCESS = 'lumen_access';
const TOKEN_KEY_REFRESH = 'lumen_refresh';

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<CurrentUser | null>(null);
  const [isLoading, setLoading] = useState(true);

  // 启动恢复：从 localStorage 读 access → 调 /auth/me → 失败则 logout
  useEffect(() => {
    (async () => {
      const access = getAccessToken();
      if (!access) { setLoading(false); return; }
      try {
        const me = await request.get<any>('/auth/me');
        setUser({ userId: Number(me.userId), tenantId: Number(me.tenantId), username: String(me.userId), roles: me.roles ?? [], perms: me.perms ?? [] });
      } catch {
        clearAuthTokens();
        setUser(null);
      } finally { setLoading(false); }
    })();
  }, []);

  const login = useCallback(async (access: string, refresh: string) => {
    setAuthTokens(access, refresh);
    const me = await request.get<any>('/auth/me');
    setUser({ userId: Number(me.userId), tenantId: Number(me.tenantId), username: String(me.userId), roles: me.roles ?? [], perms: me.perms ?? [] });
  }, []);

  const logout = useCallback(() => {
    clearAuthTokens();
    setUser(null);
  }, []);

  const refresh = useCallback(async () => {
    const oldRefresh = getRefreshToken();
    if (!oldRefresh) throw new Error('no refresh token');
    const res = await request.post<any>('/auth/refresh', { refreshToken: oldRefresh });
    setAuthTokens(res.accessToken, res.refreshToken);
    return res.accessToken;
  }, []);

  const value = useMemo<AuthContextValue>(() => ({
    user, isAuthenticated: !!user, isLoading,
    login, logout, refresh, setUser,
  }), [user, isLoading, login, logout, refresh]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}
```

```tsx
import { Navigate, useLocation } from 'react-router-dom';
import { useAuth } from './useAuth';
import { LoadingSkeleton } from '@/components/LoadingSkeleton';

export interface ProtectedRouteProps {
  children: React.ReactNode;
  perm?: string; // 可选，要求特定权限
}

export function ProtectedRoute({ children, perm }: ProtectedRouteProps) {
  const { isAuthenticated, isLoading, user } = useAuth();
  const location = useLocation();

  if (isLoading) return <LoadingSkeleton type="page" />;
  if (!isAuthenticated) {
    return <Navigate to={`/login?from=${encodeURIComponent(location.pathname + location.search)}`} replace />;
  }
  if (perm && !user!.perms.includes(perm)) {
    return <Navigate to="/403" replace />;
  }
  return <>{children}</>;
}
```

### Task 1.3: silentRefresh（axios 401 拦截器）

**Files:**

- [x] **Step 1: 写 `silentRefresh.ts`**

```ts
import { getAccessToken, setAuthTokens, clearAuthTokens } from './tokenStorage';

let inflight: Promise<string> | null = null;

export async function silentRefresh(): Promise<string> {
  if (inflight) return inflight;
  const oldRefresh = localStorage.getItem('lumen_refresh');
  if (!oldRefresh) throw new Error('no refresh token');

  inflight = (async () => {
    try {
      const resp = await fetch('/api/v1/auth/refresh', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ refreshToken: oldRefresh }),
      });
      if (!resp.ok) throw new Error(`refresh failed ${resp.status}`);
      const body = await resp.json();
      if (body.code !== 0) throw new Error(body.message || 'refresh failed');
      setAuthTokens(body.data.accessToken, body.data.refreshToken);
      return body.data.accessToken as string;
    } finally {
      inflight = null;
    }
  })();

  try {
    return await inflight;
  } catch (e) {
    clearAuthTokens();
    throw e;
  }
}

export function attachSilentRefresh(axiosInstance: typeof import('axios').default) {
  axiosInstance.interceptors.response.use(
    (resp) => resp,
    async (error) => {
      const original = error.config;
      const status = error.response?.status;
      const isAuthEndpoint = original?.url?.includes('/auth/');

      if (status === 401 && !original._retried && !isAuthEndpoint && getAccessToken()) {
        original._retried = true;
        try {
          const newAccess = await silentRefresh();
          original.headers.Authorization = `Bearer ${newAccess}`;
          return axiosInstance(original);
        } catch (refreshErr) {
          clearAuthTokens();
          window.location.href = '/login';
          return Promise.reject(refreshErr);
        }
      }
      return Promise.reject(error);
    },
  );
}
```

```tsx
import { Component, ErrorInfo, ReactNode } from 'react';
import { Button, Result } from 'antd';

export interface ErrorBoundaryProps {
  level: 'app' | 'page';
  children: ReactNode;
  onError?: (error: Error, info: ErrorInfo) => void;
}

interface State { hasError: boolean; error?: Error; }

export class ErrorBoundary extends Component<ErrorBoundaryProps, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(error: Error): State { return { hasError: true, error }; }

  componentDidCatch(error: Error, info: ErrorInfo) {
    this.props.onError?.(error, info);
    console.error('[ErrorBoundary]', this.props.level, error, info);
  }

  reset = () => this.setState({ hasError: false, error: undefined });

  render() {
    if (!this.state.hasError) return this.props.children;
    if (this.props.level === 'app') {
      return <Result status="500" title="应用异常" subTitle="请联系管理员或刷新页面重试" extra={<Button type="primary" onClick={() => location.reload()}>刷新</Button>} />;
    }
    return <Result status="warning" title="本页面加载失败" subTitle={this.state.error?.message} extra={<Button onClick={this.reset}>重试</Button>} />;
  }
}
```

```ts
import { useAuth } from '@/auth/useAuth';

export interface Access {
  perms: string[];
  roles: string[];
  canRead: (perm: string) => boolean;
  canAny: (perms: string[]) => boolean;
  canAll: (perms: string[]) => boolean;
  hasRole: (role: string) => boolean;
}

export function useAccess(): Access {
  const { user } = useAuth();
  const perms = user?.perms ?? [];
  const roles = user?.roles ?? [];
  return {
    perms, roles,
    canRead: (p) => perms.includes(p),
    canAny: (ps) => ps.some((p) => perms.includes(p)),
    canAll: (ps) => ps.every((p) => perms.includes(p)),
    hasRole: (r) => roles.includes(r),
  };
}
```

```ts
import { useCallback } from 'react';
import { PageResult } from '@/types/api';

export interface TableRequestParams {
  current?: number;
  pageSize?: number;
  [k: string]: any;
}

export interface TableRequestResult<T> {
  data: T[];
  total: number;
  success: boolean;
}

export function useTableRequest<T>(
  fetcher: (params: { pageNum: number; pageSize: number; [k: string]: any }) => Promise<PageResult<T>>,
) {
  return useCallback(async (params: TableRequestParams): Promise<TableRequestResult<T>> => {
    const { current = 1, pageSize = 20, ...rest } = params;
    try {
      const res = await fetcher({ pageNum: current, pageSize, ...rest });
      return { data: res.records ?? [], total: res.total ?? 0, success: true };
    } catch (e) {
      console.error('[useTableRequest]', e);
      return { data: [], total: 0, success: false };
    }
  }, [fetcher]);
}
```

```ts
import { useEffect, useState } from 'react';
import { request } from '@/services/request';
import type { SysDictItem } from '@/types/api';

const cache = new Map<string, Map<string, string>>();

export function useDict(code: string): Record<string, string> {
  const [map, setMap] = useState<Record<string, string>>(() => {
    const hit = cache.get(code);
    return hit ? Object.fromEntries(hit) : {};
  });

  useEffect(() => {
    if (cache.has(code)) return;
    let cancelled = false;
    request.get<SysDictItem[]>(`/dicts/${code}/items`).then((items) => {
      if (cancelled) return;
      const m = new Map<string, string>();
      (items ?? []).forEach((it) => m.set(it.code, it.label));
      cache.set(code, m);
      setMap(Object.fromEntries(m));
    });
    return () => { cancelled = true; };
  }, [code]);

  return map;
}
```

```java
public R<Void> assignPerms(@PathVariable Long id, @RequestBody List<Long> permIds) {
    Long tid = TenantContext.require();
    rolePermMapper.delete(new QueryWrapper<SysRolePermission>().eq("role_id", id).eq("tenant_id", tid));
```

```ts
import { useEffect, useCallback, useState } from 'react';
import { lightTokens, darkTokens } from './tokens';
import type { ThemeConfig } from 'antd';

const KEY = 'lumen_theme_mode';
type Mode = 'light' | 'dark';

export function useThemeMode() {
  const [mode, setMode] = useState<Mode>(() => (localStorage.getItem(KEY) as Mode) ?? 'light');

  useEffect(() => {
    localStorage.setItem(KEY, mode);
    document.documentElement.dataset.theme = mode;
  }, [mode]);

  const toggle = useCallback(() => setMode((m) => (m === 'light' ? 'dark' : 'light')), []);

  const theme: ThemeConfig = mode === 'light' ? lightTokens : darkTokens;
  return { mode, theme, toggle };
}
```

```tsx
import { createContext, useCallback, useContext, useMemo, useState } from 'react';
import { zhCN } from './zh-CN';
import { enUS } from './en-US';

export type Locale = 'zh-CN' | 'en-US';
const messages: Record<Locale, Record<string, string>> = { 'zh-CN': zhCN, 'en-US': enUS };

interface Ctx { locale: Locale; setLocale: (l: Locale) => void; t: (k: string) => string; }

const LocaleContext = createContext<Ctx | null>(null);

export function LocaleProvider({ children }: { children: React.ReactNode }) {
  const [locale, setLocale] = useState<Locale>(() => (localStorage.getItem('lumen_locale') as Locale) ?? 'zh-CN');
  const t = useCallback((k: string) => messages[locale][k] ?? k, [locale]);
  const value = useMemo(() => ({ locale, setLocale, t }), [locale, t]);
  return <LocaleContext.Provider value={value}>{children}</LocaleContext.Provider>;
}

export function useLocale(): Ctx {
  const ctx = useContext(LocaleContext);
  if (!ctx) throw new Error('useLocale outside LocaleProvider');
  return ctx;
}
```
